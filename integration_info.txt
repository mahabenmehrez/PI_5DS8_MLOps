===== agents/checker/actions.py =====
"""
One function per menu action of the Data Checker (roadmap 9.3).

Each action updates the batches table (state) and announces it on the board
(messages announce, tables hold state). They take the agent so they can use
its board, tables and settings, and return a Result for verify().
"""
from __future__ import annotations

import shutil
from datetime import timedelta
from pathlib import Path

import pandas as pd

from shared import clock
from shared.base_agent import Result
from shared.messages import MessageType

from agents.checker.analysis import BatchReport, repair_until_clean
from agents.checker.settings import node

ALL_AGENTS = ["setup", "reviewer", "trainer", "deployer", "monitor"]


def _expires(hours: float = 1.0) -> str:
    return (clock.now() + timedelta(hours=hours)).isoformat().replace("+00:00", "Z")


def announce(agent, report: BatchReport, status: str, path: str, repair: list | None = None) -> str:
    """data.batch_checked to the Trainer and the Monitor."""
    return agent.post(
        type=MessageType.FINDING, kind="data.batch_checked", to=["trainer", "monitor"],
        topic=f"dataset:{report.batch_id}", level=1, incident_id=agent.incident_id,
        summary=f"Batch {report.batch_id} {status}"
                + (f": {', '.join(report.failed_checks()[:3])}" if report.actionable else ""),
        evidence={"psi": report.psi, "isolation": report.isolation,
                  "findings": [f.to_dict() for f in report.actionable][:10]},
        payload={"batch_id": report.batch_id, "status": status, "rows": report.rows,
                 "failed_checks": report.failed_checks(), "psi": report.psi,
                 "repair": [s["repair"] for s in (repair or [])], "clean_path": path})


def pass_batch(agent, report: BatchReport, warning: bool = False) -> Result:
    batch = agent.tables.batch(report.batch_id)
    agent.tables.set_batch_status(report.batch_id, "passed",
                                  reason="pass_with_warning" if warning else "pass",
                                  clean_path=batch["path"], schema_version=agent.settings.schema_version,
                                  findings=[f.to_dict() for f in report.findings])
    agent.tables.add_batch_stats(report.batch_id, report.stats)
    announce(agent, report, "passed", batch["path"])
    return Result(ok=True, outputs={"status": "passed"})


def repair(agent, report: BatchReport, steps: list[dict]) -> Result:
    batch = agent.tables.batch(report.batch_id)
    s = agent.settings
    df = pd.read_parquet(batch["path"])
    repaired, applied, remaining = repair_until_clean(
        df, steps, agent.inspector, s.profile, s.bounds, s.thresholds["decision_min_score"])
    if remaining:                                 # the real run disagrees with the dry run
        reason = "repair failed: " + ", ".join(f"{f.check}:{f.column}" for f in remaining)
        return quarantine(agent, report, reason)
    clean = Path(s.data_dir) / "clean" / f"{report.batch_id}.parquet"
    clean.parent.mkdir(parents=True, exist_ok=True)
    repaired.to_parquet(clean, index=False)
    agent.tables.use_repaired_file(report.batch_id, str(clean))    # the Trainer reads `path`
    agent.tables.set_batch_status(report.batch_id, "repaired", reason="repair", clean_path=str(clean),
                                  schema_version=s.schema_version,
                                  findings=[f.to_dict() for f in report.findings])
    agent.tables.add_batch_stats(report.batch_id, report.stats)
    announce(agent, report, "repaired", str(clean), applied)
    return Result(ok=True, outputs={"status": "repaired", "rows_before": len(df),
                                    "rows_after": len(repaired), "steps": applied})


def quarantine(agent, report: BatchReport, reason: str) -> Result:
    batch = agent.tables.batch(report.batch_id)
    target = Path(agent.settings.data_dir) / "quarantine" / Path(batch["path"]).name
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy(batch["path"], target)
    agent.tables.set_batch_status(report.batch_id, "quarantined", reason=reason,
                                  schema_version=agent.settings.schema_version,
                                  findings=[f.to_dict() for f in report.findings])
    announce(agent, report, "quarantined", str(target))
    do_not_use(agent, [report.batch_id], reason)
    if report.live_impact:
        claim_incident(agent, report.live_impact[0].column, [report.batch_id], report.cause)
    return Result(ok=True, outputs={"status": "quarantined", "reason": reason})


def do_not_use(agent, batch_ids: list[str], reason: str) -> str:
    return agent.post(type=MessageType.REQUEST, kind="request.do_not_use", to=["trainer"],
                      topic=f"dataset:{batch_ids[0]}", level=1, incident_id=agent.incident_id,
                      summary=f"Do not train on {', '.join(batch_ids)}"[:160],
                      payload={"batch_ids": batch_ids, "reason": reason[:300]},
                      expires_at=_expires())


def claim_incident(agent, column: str, batch_ids: list[str], cause: str) -> bool:
    """incident.claim + request.hold, once per incident: the Checker owns data-caused incidents.
    Claims are kept in the claims table, so a restarted Checker does not claim twice."""
    incident = agent.incident_id
    feature = node(column)
    since = (clock.now() - timedelta(seconds=agent.window_s)).isoformat().replace("+00:00", "Z")
    if incident is None or agent.tables.recent_claim([feature], since):
        return False
    agent.tables.record_claim(incident, feature, agent.name)
    agent.post(type=MessageType.CLAIM, kind="incident.claim", to=ALL_AGENTS,
               topic=f"incident:{incident}", incident_id=incident, level=1, confidence=0.9,
               summary=f"Data incident: {cause} on {feature.split(':')[1]} since {batch_ids[0]}"[:160],
               evidence={"batches": batch_ids, "cause": cause},
               payload={"root_cause_node": feature,
                        "evidence_refs": [f"dataset:{b}" for b in batch_ids],
                        "instructions": {"trainer": "do_not_use", "deployer": "hold",
                                         "monitor": "do not request a retrain"}})
    request_hold(agent, f"{cause} on {feature} in batch {batch_ids[0]}")
    return True


def request_hold(agent, reason: str) -> str:
    return agent.post(type=MessageType.REQUEST, kind="request.hold", to=["deployer"],
                      topic="deployment:prod", level=1, incident_id=agent.incident_id,
                      summary=f"Hold releases: {reason}"[:160],
                      payload={"reason": reason,
                               "release_condition": "new batches pass schema and unit checks"},
                      expires_at=_expires())


===== agents/checker/agent.py =====
"""
Data Checker (P3): pass, repair or quarantine every incoming batch (roadmap 9).

    observe  : next `pending` row of the batches table
    analyze  : schema, statistics, nulls, ranges, unit changes, renames, duplicates,
               PSI/KS drift, Isolation Forest (analysis.py, no LLM). A clean batch is
               passed right there; anything else raises the flag.
    decide   : LLM (prompt.md) with rule_decision() as fallback and as the 'rules' system
    dry_run  : `repair_check` for repairs and schema updates (level 2)
    act      : actions.py
    handle   : query.data_changed, request.clean_window, request.expected_change

    python -m agents.checker
"""
from __future__ import annotations

import json
import logging
from datetime import timedelta
from typing import Optional

import numpy as np
import pandas as pd

from shared import clock
from shared.base_agent import Action, Analysis, BaseAgent, Decision, Result
from shared.messages import Message, MessageType
from shared.tables import Tables

from agents.checker import actions
from agents.checker.analysis import BatchReport, StatsAnomaly, analyze_batch, likely_cause
from agents.checker.checks import ALERT, DRIFT_CHECKS, Inspector, match_unit_factor
from agents.checker.repairs import REPAIRS
from agents.checker.settings import (FEATURE_COLUMNS, ROOT, CheckerSettings, column_of, node,
                                     reference_frame)
from samples.schema import make_schema

log = logging.getLogger("agents.checker")


class ClaimedIncidents:
    """Incidents that survive a restart: a problem on a feature the Checker already claimed
    (claims table in board.db) keeps that incident id; anything else asks the wrapped incidents."""

    def __init__(self, inner, tables: Tables):
        self.inner, self.tables = inner, tables

    def open_or_join(self, nodes: list[str], window_s: float) -> str:
        since = (clock.now() - timedelta(seconds=window_s)).isoformat().replace("+00:00", "Z")
        claimed = self.tables.recent_claim([n for n in nodes if n.startswith("feature:")], since)
        return claimed or self.inner.open_or_join(nodes, window_s)


class CheckerAgent(BaseAgent):
    name = "checker"
    subscriptions = ["dataset:*", "feature:*", "expected_change:*"]
    menu = [
        Action("pass", 1, success_check="registry row passed"),
        Action("pass_with_warning", 1, success_check="row passed, warning posted"),
        Action("quarantine", 1, success_check="batch absent from the next training window",
               safest=True),
        Action("repair", 2, dry_run_kind="repair_check",
               success_check="repaired copy passes all checks and PSI < 0.25 on the repaired feature"),
        Action("warn", 1, success_check="data.drift_trend or finding delivered"),
        Action("request_hold", 1, success_check="Deployer answers accepted (incident owner only)"),
        Action("update_schema", 2, dry_run_kind="repair_check",
               success_check="only after request.expected_change; next batch passes"),
    ]

    def __init__(self, tables: Optional[Tables] = None, settings: Optional[CheckerSettings] = None,
                 **kw):
        kw.setdefault("window_s", 6 * 3600)      # one incident per problem over 6 simulated hours
        super().__init__(**kw)
        self.tables = tables or Tables(str(ROOT / "board.db"))
        self.settings = settings or CheckerSettings()
        self.inspector = self._build_inspector()
        self.anomaly = StatsAnomaly(FEATURE_COLUMNS, self.settings.thresholds["isolation_min_batches"])
        self.report: Optional[BatchReport] = None
        self.incidents = ClaimedIncidents(self.incidents, self.tables)
        self.settings.save_profile()

    def _build_inspector(self) -> Inspector:
        s = self.settings
        return Inspector(s.profile, s.inspector_cfg, make_schema(s.bounds))

    # ---- 1. Look ---------------------------------------------------------------
    def observe(self) -> dict:
        pending = self.tables.pending_batches()
        if not pending:
            return {}
        batch = pending[0]
        return {"batch_id": batch["batch_id"], "path": batch["path"],
                "df": pd.read_parquet(batch["path"])}

    # ---- 2. Understand ---------------------------------------------------------
    def analyze(self, obs: dict) -> Analysis:
        self.report = None
        if not obs:
            return Analysis(flag=False)
        normal = [b["batch_id"] for b in self.tables.batches("passed") if b["reason"] == "pass"]
        self.anomaly.fit(self.tables.batch_stats(), normal)
        s = self.settings
        report = analyze_batch(obs["batch_id"], obs["df"], self.inspector, s.profile, s.bounds,
                               FEATURE_COLUMNS, s.thresholds["decision_min_score"], self.anomaly)
        self.report = report
        if not report.actionable:      # nothing that matters for the model: pass without the LLM
            actions.pass_batch(self, report)
            return Analysis(flag=False)
        nodes = [f"dataset:{report.batch_id}"] + sorted({node(f.column) for f in report.actionable
                                                         if f.column in FEATURE_COLUMNS})
        return Analysis(flag=True, nodes=nodes,
                        numbers={"rows": report.rows, "failed_checks": report.failed_checks(),
                                 "psi": report.psi, "isolation": report.isolation},
                        problem=f"{report.cause} " + " ".join(report.failed_checks()))

    # ---- 3. Decide ---------------------------------------------------------------
    def context(self, analysis: Analysis, inbox: list[Message]) -> dict:
        """What the LLM sees (prompt.md): numbers only, never rows."""
        r = self.report
        ranked = sorted(self.settings.importance.items(), key=lambda kv: -kv[1])
        return {
            "batch_id": r.batch_id, "rows": r.rows, "likely_cause": r.cause,
            "findings": [f.to_dict() for f in sorted(r.actionable, key=lambda f: -f.score)][:10],
            "psi_with_importance_rank": {c: {"psi": v, "rank": [k for k, _ in ranked].index(c) + 1}
                                         for c, v in r.psi.items() if c in self.settings.importance},
            "isolation_forest": r.isolation,
            "candidate_repairs": r.steps,
            "unfixable": [f.to_dict() for f in r.unfixable],
            "live_impact": [f"{f.check}:{f.column}" for f in r.live_impact],
            "similar_past_cases": self.memory.similar(analysis.problem, agent=self.name, k=3),
            "inbox": [{"kind": m.kind, "sender": m.sender, "summary": m.summary} for m in inbox][:5],
        }

    def rule_decision(self, analysis: Analysis) -> Decision:
        """Fallback when the LLM fails, and the whole decision in the 'rules' system.

        - an error that also reaches the live API (unit change, missing column, type) or that
          no repair can fix -> quarantine (+ claim and hold for the live ones);
        - other errors (duplicates, a few bad rows, a small null spike, rename) -> repair;
        - drift, anomalies, new columns, minor warnings -> pass with a warning."""
        r = self.report
        if r.unfixable or r.live_impact:
            blocking = r.live_impact or r.unfixable
            reason = f"{r.cause}: " + ", ".join(f"{f.check}:{f.column}" for f in blocking)
            return Decision("quarantine", 1, {"reason": reason}, reason=reason,
                            success_check="batch absent from the next training window")
        if r.errors:
            return Decision("repair", 2, self.repair_params(r.steps),
                            reason="repairable: " + ", ".join(s["repair"] for s in r.steps),
                            success_check="repaired copy passes all checks")
        if r.drift_only:
            return Decision("warn", 1, {}, reason="valid shift: " + ", ".join(r.failed_checks()),
                            success_check="data.drift_trend delivered")
        return Decision("pass_with_warning", 1, {}, reason="warnings only: " + ", ".join(r.failed_checks()))

    def repair_params(self, steps: list[dict]) -> dict:
        batch = self.tables.batch(self.report.batch_id)
        return {"batch_id": self.report.batch_id, "path": batch["path"], "steps": steps,
                "profile_path": str(self.settings.profile_path)}

    def decide(self, analysis: Analysis) -> Decision:
        """The LLM chooses the action; the agent fills in the technical parameters it cannot
        know (batch path, profile version). Repair steps come from the analysis unless the
        LLM gave valid ones."""
        decision = super().decide(analysis)
        if decision.action == "repair":
            steps = decision.params.get("steps")
            valid = isinstance(steps, list) and steps and all(
                isinstance(s, dict) and s.get("repair") in REPAIRS for s in steps)
            decision.params = self.repair_params(steps if valid else self.report.steps)
        return decision

    # ---- 5. Do -------------------------------------------------------------------
    def act(self, decision: Decision) -> Result:
        r, p = self.report, decision.params
        if decision.action == "pass":
            return actions.pass_batch(self, r)
        if decision.action == "pass_with_warning":
            return actions.pass_batch(self, r, warning=True)
        if decision.action == "quarantine":
            return actions.quarantine(self, r, p.get("reason") or decision.reason or r.cause)
        if decision.action == "repair":
            return actions.repair(self, r, p.get("steps") or r.steps)
        if decision.action == "warn":
            result = actions.pass_batch(self, r, warning=True)
            self.post_drift_trend(r)
            return result
        if decision.action == "request_hold":
            actions.request_hold(self, decision.reason or r.cause)
            return actions.quarantine(self, r, decision.reason or r.cause)
        return Result(ok=False, detail=f"{decision.action} is only used on request.expected_change")

    # ---- 6. Check ----------------------------------------------------------------
    def verify(self, decision: Decision, result: Result) -> bool:
        expected = {"quarantine": "quarantined", "request_hold": "quarantined",
                    "repair": "repaired"}.get(decision.action, "passed")
        row = self.tables.batch(self.report.batch_id)
        # a repair that failed for real ends as a quarantine: safe, but the action did not work
        return result.ok and row is not None and row["status"] == expected

    # ---- questions and requests from the others --------------------------------
    def handle(self, msg: Message) -> None:
        if msg.type.value == "answer":
            return
        handler = {"query.data_changed": self.answer_data_changed,
                   "request.clean_window": self.answer_clean_window,
                   "request.expected_change": self.answer_expected_change}.get(msg.kind)
        if handler:
            handler(msg)
        else:
            log.debug("checker ignores %s from %s", msg.kind, msg.sender)

    def answer_data_changed(self, msg: Message) -> None:
        """Compare the window's batches with the batches before it, for one feature."""
        column = column_of(msg.payload.get("feature", msg.topic))
        start, end = msg.payload.get("start") or "", msg.payload.get("end") or "￿"
        checked = [b for b in self.tables.batches() if b["status"] != "pending"]
        window = [b for b in checked if start <= (b["created_at"] or "") <= end] or checked[-1:]
        first = checked.index(window[0]) if window else 0
        before = checked[max(0, first - self.settings.thresholds["compare_last_batches"]):first]

        hits, checks = [], []
        for b in window:
            serious = [f["check"] for f in json.loads(b["findings"] or "[]")
                       if f["column"] == column and f["level"] != "warn"]
            if serious:
                hits.append(b["batch_id"])
                checks += serious
        ratio = self.median_ratio(column, [b["batch_id"] for b in window], [b["batch_id"] for b in before])
        cause = likely_cause(checks)
        if cause == "none" and ratio and match_unit_factor(ratio, 0.3, self.settings.thresholds["unit_factors"]):
            cause = "unit_change"
        changed = bool(hits) or cause != "none"

        if changed and cause in ("unit_change", "schema_change") and msg.incident_id:
            self.incident_id = msg.incident_id
            batch_ids = hits or [b["batch_id"] for b in window]
            if actions.claim_incident(self, column, batch_ids, cause):
                usable = [b for b in batch_ids if self.tables.batch(b)["status"] != "quarantined"]
                if usable:                        # quarantined batches were announced already
                    actions.do_not_use(self, usable, f"{cause} on {column}")
        self.reply(msg, level=1, confidence=0.9 if hits else 0.6,
                          summary=(f"Yes: {cause} on {column} in {', '.join(hits[:3])}" if changed
                                   else f"No change found on {column}")[:160],
                          evidence={"median_ratio_vs_before": ratio, "window_batches": len(window),
                                    "batches_before": len(before)},
                          payload={"changed": changed, "batch_ids": hits, "likely_cause": cause})

    def median_ratio(self, column: str, window: list[str], before: list[str]) -> Optional[float]:
        stats = self.tables.batch_stats()
        stats = stats[stats.column_name == column]
        now, old = stats[stats.batch_id.isin(window)]["median"], stats[stats.batch_id.isin(before)]["median"]
        if now.empty or old.empty or not old.median():
            return None
        return round(float(now.median() / old.median()), 4)

    def answer_clean_window(self, msg: Message) -> None:
        start, end = msg.payload.get("start") or "", msg.payload.get("end") or "￿"
        min_rows = int(msg.payload.get("min_rows", 0))
        clean = [b for b in self.tables.batches() if b["status"] in ("passed", "repaired")
                 and start <= (b["created_at"] or "") <= end]
        rows = sum(b["rows"] or 0 for b in clean)
        self.reply(msg, level=1,
                          summary=f"{len(clean)} clean batches, {rows} rows",
                          payload={"decision": "done" if rows >= min_rows else "failed",
                                   "detail": "" if rows >= min_rows else f"only {rows} rows < {min_rows}",
                                   "batch_ids": [b["batch_id"] for b in clean], "rows": rows})

    def answer_expected_change(self, msg: Message) -> None:
        """Migration plan from the Reviewer: new schema version for the changed feature,
        checked in a dry run first, then answer done (scenario B)."""
        tasks = msg.payload.get("tasks", {})
        if self.name not in tasks:
            return
        s = self.settings
        profile, bounds = s.profile, dict(s.bounds)
        for change in msg.payload.get("changes", []):
            profile, bounds = scale_column(profile, bounds, column_of(change["name"]),
                                           float(change["factor"]))
        ok, detail = self.check_new_schema(profile, bounds, msg.payload.get("changes", []))
        if ok:
            s.profile, s.bounds, s.schema_version = profile, bounds, s.schema_version + 1
            self.inspector = self._build_inspector()
            s.save_profile()
        self.reply(msg, level=2,
                          summary=(f"Schema v{s.schema_version} ready" if ok
                                   else f"Schema update failed: {detail}")[:160],
                          evidence={"schema_version": s.schema_version, "detail": detail},
                          payload={"decision": "done" if ok else "failed", "detail": detail,
                                   "task": tasks[self.name], "done_at": clock.iso()})

    def check_new_schema(self, profile, bounds, changes) -> tuple[bool, str]:
        """Dry run of update_schema: reference rows converted to the new unit must pass."""
        sample = reference_frame()[FEATURE_COLUMNS].copy()     # the reference, in the new unit
        for change in changes:
            c = column_of(change["name"])
            sample[c] = sample[c] * float(change["factor"])
        trial = Inspector(profile, self.settings.inspector_cfg, make_schema(bounds))
        blocking = [f for f in trial.inspect(sample) if f.level >= ALERT and f.check not in DRIFT_CHECKS]
        return (not blocking, "; ".join(f"{f.check}:{f.column}" for f in blocking))

    def post_drift_trend(self, r: BatchReport) -> None:
        stats = self.tables.batch_stats()
        for column, value in r.psi.items():
            history = stats[stats.column_name == column]["median"].tail(10).to_numpy()
            slope = float(np.polyfit(range(len(history)), history, 1)[0]) if len(history) > 2 else 0.0
            self.post(type=MessageType.FINDING,
                      kind="data.drift_trend", to=["monitor", "trainer"], topic=node(column), level=1,
                      summary=f"Drift on {column}: PSI {value}", evidence={"psi": value},
                      payload={"feature": node(column).split(":")[1], "statistic": "median",
                               "slope_per_batch": round(slope, 5), "batches_to_limit": None})


def scale_column(profile: dict, bounds: dict, column: str, factor: float):
    """New reference after an approved unit change: the same distribution, times factor."""
    import copy
    profile, bounds = copy.deepcopy(profile), dict(bounds)
    ref = profile["columns"][column]
    for key in ("lo", "hi", "median", "p25", "p75"):
        ref[key] = ref[key] * factor
    ref["edges"] = [e * factor for e in ref["edges"]]
    ref["sample"] = [v * factor for v in ref.get("sample", [])]
    if column in bounds:
        lo, hi = bounds[column]
        bounds[column] = (lo * factor, hi * factor)
    return profile, bounds



===== agents/checker/analysis.py =====
"""
Statistics and checks of the Data Checker (no LLM here, roadmap 5.2).

analyze_batch() turns one batch into a BatchReport: findings (schema, nulls,
ranges, unit changes, renames, duplicates, PSI/KS drift), the batch statistics
for the batch_stats table, an Isolation Forest score on those statistics, the
likely cause and the repair plan.
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Optional

import numpy as np
import pandas as pd
from sklearn.ensemble import IsolationForest

from agents.checker.checks import ALERT, DRIFT_CHECKS, WARN, Finding, Inspector
from agents.checker.repairs import apply_repairs, plan_repairs

# Errors in the feed that also reach the live API (the same values go to /predict):
# the incident is data-caused, so the Checker claims it and asks for a hold (scenario C).
LIVE_IMPACT_CHECKS = {"unit_change", "missing_column", "type_changed"}

CAUSES = {                       # finding -> likely_cause of the query.data_changed answer
    "unit_change": "unit_change",
    "missing_column": "schema_change", "renamed_column": "schema_change",
    "new_column": "schema_change", "type_changed": "schema_change",
    "schema_range": "schema_change", "out_of_range": "schema_change",
    "null_spike": "missing_values",
    "empty_batch": "late_batch",
    "drift": "valid_shift", "range_shift": "valid_shift", "batch_anomaly": "valid_shift",
}
CAUSE_PRIORITY = ["unit_change", "schema_change", "missing_values", "late_batch", "valid_shift"]


@dataclass
class BatchReport:
    batch_id: str
    rows: int
    findings: list[Finding]
    actionable: list[Finding]
    steps: list[dict]
    unfixable: list[Finding]
    stats: list[dict]
    psi: dict = field(default_factory=dict)
    isolation: Optional[dict] = None

    @property
    def drift_only(self) -> bool:
        return bool(self.actionable) and all(f.check in DRIFT_CHECKS for f in self.actionable)

    @property
    def errors(self) -> list[Finding]:
        return [f for f in self.actionable if f.level >= ALERT and f.check not in DRIFT_CHECKS]

    @property
    def live_impact(self) -> list[Finding]:
        """Errors that also reach the live API: a unit or type change, a missing column, or a
        model feature invalid on most rows."""
        return [f for f in self.errors if f.check in LIVE_IMPACT_CHECKS
                or (f.check == "schema_range" and f.evidence.get("rate", 0) >= 0.5)]

    @property
    def cause(self) -> str:
        return likely_cause(self.actionable)

    def failed_checks(self) -> list[str]:
        return [f"{f.check}:{f.column}" if f.column else f.check for f in self.actionable]


def likely_cause(findings: list) -> str:
    """Findings or plain check names -> the most specific cause."""
    found = {CAUSES.get(getattr(f, "check", f)) for f in findings} - {None}
    return next((c for c in CAUSE_PRIORITY if c in found), "none")


def batch_stats(df: pd.DataFrame, columns: list[str]) -> list[dict]:
    out = []
    for c in columns:
        if c not in df.columns:
            continue
        s = pd.to_numeric(df[c], errors="coerce")
        v = s.dropna()
        q = v.quantile([0.01, 0.5, 0.99]) if len(v) else pd.Series([np.nan] * 3, index=[0.01, 0.5, 0.99])
        out.append({"column": c, "null_rate": float(s.isna().mean()),
                    "mean": float(v.mean()) if len(v) else None,
                    "median": float(q[0.5]), "p1": float(q[0.01]), "p99": float(q[0.99])})
    return out


class StatsAnomaly:
    """Isolation Forest on the statistics vector of a batch (median and null rate per feature).

    Trained once on the first N normal batches. A batch is an anomaly when it scores lower
    than every training batch, which keeps false alarms close to zero on normal days."""

    def __init__(self, columns: list[str], min_batches: int = 10):
        self.columns = columns
        self.min_batches = min_batches
        self.model: Optional[IsolationForest] = None
        self.floor = None

    def vector(self, stats: list[dict]) -> list[float]:
        by_col = {s["column"]: s for s in stats}
        return [by_col.get(c, {}).get(k) or 0.0 for c in self.columns for k in ("median", "null_rate")]

    def fit(self, stats_table: pd.DataFrame, normal_batches: list[str]) -> bool:
        if self.model is not None or len(normal_batches) < self.min_batches:
            return self.model is not None
        train = []
        for b in normal_batches[: self.min_batches]:
            rows = stats_table[stats_table.batch_id == b].rename(columns={"column_name": "column"})
            train.append(self.vector(rows.to_dict("records")))
        self.model = IsolationForest(n_estimators=200, random_state=0).fit(train)
        self.floor = float(self.model.score_samples(train).min())
        return True

    def score(self, stats: list[dict]) -> Optional[dict]:
        if self.model is None:
            return None
        s = float(self.model.score_samples([self.vector(stats)])[0])
        return {"score": round(s, 4), "floor": round(self.floor, 4), "anomaly": bool(s < self.floor - 0.02)}


def analyze_batch(batch_id: str, df: pd.DataFrame, inspector: Inspector, profile: dict,
                  bounds: dict, columns: list[str], min_score: float,
                  anomaly: Optional[StatsAnomaly] = None) -> BatchReport:
    findings = inspector.inspect(df)
    stats = batch_stats(df, columns)
    iso = anomaly.score(stats) if anomaly else None
    if iso and iso["anomaly"]:
        findings.append(Finding("batch_anomaly", None, WARN, 1.0, iso))
    # Importance decides how serious a finding is, but a significant drift is always worth a
    # warning, even on a feature the model barely uses (scenario E: warn, never quarantine).
    actionable = [f for f in findings if f.score >= min_score
                  or (f.check in DRIFT_CHECKS and f.level >= ALERT)]
    steps, unfixable = plan_repairs(actionable, profile, bounds)
    psi = {f.column: f.evidence["psi"] for f in findings if f.check == "drift"}
    return BatchReport(batch_id, len(df), findings, actionable, steps, unfixable, stats, psi, iso)


def repair_until_clean(df: pd.DataFrame, steps: list[dict], inspector: Inspector, profile: dict,
                       bounds: dict, min_score: float, max_rounds: int = 3):
    """Apply, re-inspect, plan again. Returns (df, steps applied, remaining blocking findings)."""
    applied, remaining = [], []
    for _ in range(max_rounds):
        df = apply_repairs(df, steps)
        applied += steps
        remaining = [f for f in inspector.inspect(df)
                     if f.score >= min_score and f.level >= ALERT and f.check not in DRIFT_CHECKS]
        if not remaining:
            return df, applied, []
        steps, unfixable = plan_repairs(remaining, profile, bounds)
        if unfixable or not steps:
            break
    return df, applied, remaining

    # 4. Try first: the dry run, in a subprocess with DRY_RUN=1 (roadmap 5.8)
    def dry_run(self, kind: str, inputs: Dict[str, Any]) -> Dict[str, Any]:
        with tempfile.NamedTemporaryFile("w", suffix=".json", delete=False, encoding="utf-8") as f:
            json.dump(inputs, f)
        try:
            r = subprocess.run([sys.executable, str(HERE.parents[1] / "dry_run_jobs" / f"{kind}.py"), f.name],
                               capture_output=True, text=True, env={**os.environ, "DRY_RUN": "1"},

        finally:
            os.unlink(f.name)

    @staticmethod
    def fix_for(result: Dict[str, Any], plan, env: Optional[Dict[str, Any]] = None) -> Optional[str]:
        """apply_fix (level 2): the fixes the rules know. Changes `plan` (or `env`, the dry run's settings) and
        returns what was changed; None when no rule applies (then: give_up_and_report)."""
        from agents.setup.builder_agent.scan.known import IMPORT_TO_DIST
        env = {} if env is None else env
        if result["error_class"] == "BrokenEnvironment" and not env.get("fresh_venv"):
            env["fresh_venv"] = True                # once: a second broken install is reported, not retried
            return "recreate the dry-run environment (an install was broken)"
        if result["error_class"] == "ModuleNotFoundError" and result.get("detail"):
            top = result["detail"].split(".")[0]
            root = Path(plan.layout.root)
            local = [root / top, root / f"{top}.py", root / plan.layout.code_dir / top,
                     root / plan.layout.code_dir / f"{top}.py"]
            if any(p.exists() for p in local):
                return None     # the project's own module: a path problem, never a pip install of a same-named package
            dist = IMPORT_TO_DIST.get(top, top)
            if dist not in plan.extra_packages:
                plan.extra_packages.append(dist)
                env["action"] = {"action": "add_dependency", "name": dist}
                return f"install {dist} (the code imports {top})"
        log = result.get("log", "")
        script = getattr(plan, "train_script", None)
        if result.get("stage") == "train" and script:
            from agents.setup.builder_agent.team import module_of
            root = Path(plan.layout.root)
            # change_entry_command: a relative import only works when the script runs as a module of its package
            if ("attempted relative import with no known parent package" in log and plan.train_style == "script"
                    and module_of(script, root)):
                plan.train_style = "module"
                env["action"] = {"action": "change_entry_command", "stage": "train", "style": "module"}
                return f"run train as a module (python -m {module_of(script, root)[1]}): it uses relative imports"
            # fix_path: the missing relative path exists from the script's folder or one of its parents (up to the
            # repo root): the script expects to run from there. The nearest folder wins.
            if result["error_class"] == "FileNotFoundError" and result.get("detail"):
                from agents.setup.builder_agent.team.fix import current_cwd
                missing, here = result["detail"], current_cwd(plan)
                if not Path(missing).is_absolute() and not (root / here / missing).exists():
                    folder = Path(script).parent
                    while True:
                        f = folder.as_posix() or "."
                        if f != here and (root / f / missing).exists():
                            plan.train_cwd = f
                            env["action"] = {"action": "fix_path", "target": "workdir", "path": f}
                            where = "next to the script" if f == (Path(script).parent.as_posix() or ".") \
                                else f"in {f}/, above the script"
                            return f"run train from {f}/ ({missing} is there, {where})"
                        if f == ".":
                            break
                        folder = folder.parent
        return None

    def act(self, decision: Decision) -> Dict[str, Any]:

            raise ValueError(f"unknown action {decision.action}")
        from agents.setup.builder_agent.team import TeamLayout, render_team, team_plan, write_team
        layout = TeamLayout.detect(self.root)     # the team layout, or a small repo's (the test repos)
        plan = team_plan(layout)
        # the decide step: choices the rules couldn't settle (needs_llm) go to the LLM, the rest stay as decided
        from agents.setup.builder_agent.team import resolve_with_llm
        needs = any(d["status"] == "needs_llm" for d in plan.decisions)
        plan, decide_log = resolve_with_llm(plan, self.llm() if needs else None)
        th = self.thresholds()
        plan.sample_rows = int(th.get("dry_run_sample_rows", plan.sample_rows))
        max_fixes, timeout_s = int(th.get("max_fix_attempts", 3)), int(th.get("dry_run_timeout_s", 600))
        from agents.setup.builder_agent.team.fix import apply, diagnose_team, regressed
        fixes: List[str] = []                  # what was applied, in order (reverted ones say so)
        attempts: List[Dict[str, Any]] = []    # one per applied fix, for the memory write-back
        diagnoses: List[Dict[str, Any]] = []   # what the LLM was asked and answered
        reverted: List[Dict[str, Any]] = []
        tried: List[Dict[str, Any]] = []       # every fix applied (rules and LLM): never proposed twice
        escalated = False                      # cascade: the escalation model has taken over
        env: Dict[str, Any] = {"fresh_venv": False}
        rebuild = decision.params.get("rebuild", False)
        from agents.setup.builder_agent.render import MARKER

        def run() -> Dict[str, Any]:
            return self.dry_run("build", {"root": self.root, "files": render_team(plan), "timeout_s": timeout_s,
                                          "makefile": layout.makefile, "marker": MARKER,
                                          "update_own": [layout.makefile, layout.ci] if rebuild else [], **env})

        def failed(result, why: str) -> Dict[str, Any]:
            return {"ok": False, "result": result, "fixes": fixes, "plan": plan, "why": why, "diagnoses": diagnoses,
                    "attempts": attempts, "decide_log": decide_log}

        result = run()
        while not result["ok"]:
            if len(fixes) >= max_fixes:
                return failed(result, f"{max_fixes} fixes tried")
            rule = self.fix_for(result, plan, env)
            if rule is not None:
                # stored in memory as the menu action it amounts to; recreating the venv is not a project fix
                action = env.pop("action", None)   # recreating the venv is not a project fix: not remembered
                if action is not None:
                    tried.append(action)
                trial, source, text, reason, retrieval = run(), "rule", rule, "rule", None
            else:
                second = self.escalation_llm()
                client = second if escalated else self.diagnosis_llm()
                if client is None:
                    return failed(result, "no rule fixes this error and no LLM is configured")
                crag = self.crag()

                def ask(model):
                    return diagnose_team(plan, result, model, fixes, len(fixes) + 1, max_fixes,
                                         memory=self.memory(), reverted=reverted, crag=crag,
                                         scan_summary=self._scan_summary(plan), tried=tried)
                d, fix = ask(client)
                diagnoses.append(d.model_dump())
                if fix is None and second is not None and not escalated:
                    # cascade: the first model gave up or gave no valid answer: the second one, same attempt
                    escalated = True
                    d, fix = ask(second)
                    diagnoses.append(d.model_dump() | {"escalated": True})
                elif escalated:
                    diagnoses[-1]["escalated"] = True
                if fix is None:
                    return failed(result, f"LLM: {d.text}" + (f" ({d.reason})" if d.reason else ""))
                before = (list(plan.extra_packages), plan.train_cwd, plan.train_style)
                tried.append(fix.model_dump())
                apply(plan, fix)
                trial, source, text, action, reason = run(), "llm", d.text, d.fix, d.reason
                retrieval = d.crag or {"action": "off", "memory_cases": d.retrieved}
                stuck = (trial["stage"], trial.get("error_class")) == (result["stage"], result.get("error_class"))
                if not trial["ok"] and second is not None and (stuck or regressed(result, trial)):
                    escalated = True                 # cascade: its fix changed nothing or broke more: the second model goes on
                if regressed(result, trial):
                    plan.extra_packages[:], plan.train_cwd, plan.train_style = before
                    reverted.append(d.fix)
                    fixes.append(f"{text}: reverted (it broke the earlier stage {trial['stage']})")
                    attempts.append(self._attempt(len(fixes), result, action, text, reason, source,
                                                  f"reverted: regression (failed at {trial['stage']})", retrieval))
                    continue
            fixes.append(text)
            if action is not None:
                attempts.append(self._attempt(len(fixes), result, action, text, reason, source,
                                              "passed" if trial["ok"] else f"failed at {trial['stage']}", retrieval))
            result = trial
        files = render_team(plan)
        from agents.setup.builder_agent.repo_writer import current_branch, is_git_repo
        start_branch = current_branch(Path(self.root)) if is_git_repo(Path(self.root)) else None
        report = write_team(plan, files, self.root, update_own=rebuild)
        sha = self._commit(report.created + report.updated, decision.reason, start_branch) if self.commit else None
        return {"ok": True, "result": result, "fixes": fixes, "plan": plan, "report": report, "commit": sha,
                "files": files, "diagnoses": diagnoses, "attempts": attempts, "decide_log": decide_log}

    @staticmethod
    def _attempt(n, result, action, text, reason, source, outcome, retrieval=None) -> Dict[str, Any]:
        """retrieval: what the LLM was shown (CRAG action, kept memory cases, web query and URLs), so a learned
        fix says where its knowledge came from; None for a rule fix."""
        return {"n": n, "stage": result["stage"], "error_tail": result.get("log", "").splitlines()[-50:],
                "action": action, "action_text": text, "reason": reason, "source": source, "outcome": outcome,
                "retrieval": retrieval,
                "sandbox_attempt_id": f"team-{datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%S%f')}-{n}"}

    @staticmethod
    def _scan_summary(plan) -> str:
        return "\n".join([f"code: {plan.layout.code_dir}/, Python {plan.python_version}",
                          f"train: {plan.train_script}, serve: {plan.serve_target}",
                          f"data: {plan.layout.data_path} (separator {plan.separator!r}, target {plan.target})",
                          *(f"finding: {f.message}" for f in plan.findings)])

    def _commit(self, created: List[str], reason: str, start_branch: Optional[str]) -> Optional[str]:
        """Commit the created files on builder/<date>, then go back to the branch the repo was on: the commit
        waits on its branch for the Reviewer (roadmap 7: after go-live, a branch the Reviewer reviews)."""
        from agents.setup.builder_agent.decide.contracts import load_contracts
        from agents.setup.builder_agent.repo_writer import RepoWriter, current_branch, is_git_repo
        root = Path(self.root).resolve()
        if not created or not is_git_repo(root):
            return None
        writer = RepoWriter(root)                    # already on builder/<date>: write_team switched there
        writer.created = set(created)
        author = load_contracts(HERE / "contracts.yaml").i("reviewer", "builder_commit_author")
        branch = current_branch(root)
        sha = writer.commit(f"Setup Agent: pipeline files ({reason})"[:72] + f"\n\nOn {branch}.", author)
        if start_branch and start_branch != branch:
            writer.restore(start_branch)
        self.branch = branch
        return sha

    # 6. Check, then post the result
    def verify(self, decision: Decision, out: Dict[str, Any]) -> Message:
        plan = out["plan"]
        lay = plan.layout
        if out["ok"]:
            payload = {"repo": lay.code_dir, "commit": out["commit"] or "", "files": {
                "Makefile": lay.makefile, "ci": lay.ci, "schema": plan.team_schema or lay.schema_file,
                "agents_yaml": lay.agents_yaml,
                "dependency_map": lay.depmap}, "agents_yaml": lay.agents_yaml,
                "schema_path": plan.team_schema or lay.schema_file,
                "depmap_seeded": True, "branch": self.branch}
            created = out["report"].created
            msg = Message(type=MessageType.FINDING, kind="pipeline.setup_completed", sender=self.name, to=[],
                          topic=TOPIC, level=1, confidence=1.0,
                          summary=(f"Pipeline ready for {lay.code_dir}: make all passes on a sample, "
                                   f"{len(created)} files created, {len(out['fixes'])} fixes")[:160],
                          evidence={"dry_run_seconds": out["result"].get("seconds"), "fix_attempts": len(out["fixes"]),
                                    "fixes": out["fixes"], "diagnoses": _brief(out.get("diagnoses", [])), "created": created,
                                    "updated": out["report"].updated,
                                    "decisions": _decisions(plan.decisions), "decide_calls": len(out.get("decide_log") or []),
                                    "verified": out["report"].verified,
                                    "findings": [f.model_dump() for f in plan.findings]},
                          payload=payload)
        else:
            r = out["result"]
            logs = Path(self.root) / LOGS
            logs.mkdir(parents=True, exist_ok=True)
            log_ref = (LOGS / f"{datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%S')}-{r['stage']}.log").as_posix()
            (Path(self.root) / log_ref).write_text(r.get("log", ""), encoding="utf-8")
            msg = Message(type=MessageType.FINDING, kind="pipeline.setup_failed", sender=self.name, to=["human"],
                          topic=TOPIC, level=1, confidence=1.0,
                          summary=(f"Setup failed at {r['stage']} ({r['error_class']}) after "
                                   f"{len(out['fixes'])} fixes")[:160],
                          evidence={"detail": r.get("detail"), "fixes": out["fixes"], "why": out.get("why"),
                                    "decisions": _decisions(plan.decisions), "decide_calls": len(out.get("decide_log") or []),
                                    "diagnoses": _brief(out.get("diagnoses", []))},
                          payload={"stage": r["stage"], "error_class": r["error_class"],
                                   "attempts": len(out["fixes"]),
                                   "log_ref": log_ref})
        self.outbox.append(msg)
        return msg

    # 7. Remember: every fix the dry run checked goes to memory/learned/ with its outcome; retrieval only
    # uses the ones that passed (builder_agent.memory.learned: "learned" = base + passed fixes)
    def remember(self, out: Dict[str, Any]) -> None:
        verified = [a["action_text"] for a in out.get("attempts", []) if a["outcome"] == "passed"]
        self.history.append({"ok": out["ok"], "fixes": out["fixes"] if out["ok"] else [], "verified": verified,
                             "error_class": None if out["ok"] else out["result"]["error_class"]})
        if not (self.learn and out.get("attempts")):
            return
        from types import SimpleNamespace

        from agents.setup.builder_agent.memory.learned import write_back
        client = self._diagnosis_client if self._diagnosis_client not in ("auto", None) else None
        from agents.setup.builder_agent.team.fix import PROMPT
        crag_on = self.use_crag is not False
        run = SimpleNamespace(repo=out["plan"].layout.code_dir, model=getattr(client, "model", None),
                              model_digest=getattr(client, "digest", None),
                              prompt_version=PROMPT,
                              retriever="crag" if crag_on else "advanced",
                              attempts=[SimpleNamespace(**a) for a in out["attempts"]])
        write_back(run, memory_set="learned", directory=self.memory_dir)
        if self._memory not in ("auto", None):
            self._memory = "auto"                    # rebuilt on next use, with what was just learned

    def _log(self, decision: "Decision", out: Dict[str, Any], msg: Message) -> None:
        """Every LLM call, action, check and outcome is logged (roadmap 13.1.6): agents/setup/logs/runs.jsonl
        (one line per step) and llm_calls.jsonl (one line per call, prompts and answers in full)."""
        logs = Path(self.root) / LOGS.parent
        logs.mkdir(parents=True, exist_ok=True)
        now = datetime.now(timezone.utc).isoformat(timespec="seconds")
        run = {"time": now, "message_id": msg.id, "kind": msg.kind, "decision": decision.action,
               "reason": decision.reason, "level": decision.level, "fixes": out.get("fixes", []),
               "attempts": [{k: a[k] for k in ("n", "stage", "action_text", "source", "outcome")}
                            for a in out.get("attempts", [])],
               "checks": out.get("checks", {}), "dry_run_seconds": out["result"].get("seconds"),
               "why": out.get("why"), "decisions": _decisions(out["plan"].decisions)}
        with open(logs / "runs.jsonl", "a", encoding="utf-8") as f:
            f.write(json.dumps(run, ensure_ascii=False) + "\n")
        with open(logs / "llm_calls.jsonl", "a", encoding="utf-8") as f:
            for c in out.get("decide_log", []):
                f.write(json.dumps({"time": now, "message_id": msg.id, "step": "decide", **c},
                                   ensure_ascii=False) + "\n")
            for n, d in enumerate(out.get("diagnoses", []), 1):
                for c in d.get("call_log", []):
                    f.write(json.dumps({"time": now, "message_id": msg.id, "diagnosis": n, "model": d.get("model"),
                                        "crag_action": (d.get("crag") or {}).get("action"), **c},
                                       ensure_ascii=False) + "\n")

    def compose_check(self) -> str:
        """write_files' success check (roadmap 7.3): docker compose config passes on the stack's compose file."""
        import shutil
        files = [p for p in (Path(self.root) / self.repo / "docker-compose.yaml",
                             Path(self.root) / self.repo / "docker-compose.yml") if p.is_file()]
        if not files:
            return "no compose file"
        if shutil.which("docker") is None:
            return "skipped (docker not installed here)"
        r = subprocess.run(["docker", "compose", "-f", str(files[0]), "config", "-q"], capture_output=True,
                           text=True, timeout=60)
        return "passes" if r.returncode == 0 else f"fails: {r.stderr.strip()[:300]}"

    def step(self) -> Optional[Message]:
        """One turn of the loop. Returns the result message, or None when there is nothing to do."""
        analysis = self.analyze(self.observe())
        if not analysis.flag:
            return None
        decision = self.rule_decision(analysis)
        out = self.act(decision)
        out["checks"] = {"docker_compose_config": self.compose_check()}
        msg = self.verify(decision, out)
        msg.evidence["checks"] = out["checks"]
        self.remember(out)
        self._log(decision, out, msg)
        if analysis.trigger is not None:
            msg.reply_to = analysis.trigger.id
            msg.incident_id = analysis.trigger.incident_id
            self.mark_handled(analysis.trigger)
        self.first_run = False
        return msg

    # Messages from others
    def handle(self, msg: Message) -> Optional[Message]:
        """request.rebuild -> "accepted" answer (the work happens in the next step()). Idempotent."""
        if msg.id in self._handled:
            return None
        if msg.kind != "request.rebuild":
            print(f"[setup] ignoring unknown kind {msg.kind}")
            return None
        self.inbox.append(msg)
        answer = Message(
            type=MessageType.ANSWER, kind="request.rebuild", sender=self.name, to=[msg.sender],
            topic=TOPIC, reply_to=msg.id, incident_id=msg.incident_id, level=1,
            summary=f"Rebuild accepted: {msg.payload.get('reason', '')}"[:160],
            payload={"status": "accepted", "changed_paths": msg.payload.get("changed_paths", [])},
        )
        self.outbox.append(answer)
        return answer

    def mark_handled(self, msg: Message) -> None:
        self._handled.add(msg.id)

"""
Stub (fake) Setup Agent (P1) for teammates, weeks 1 and 2 (roadmap 5.1 rule 5, 7.6).

Posts pipeline.setup_completed pointing at the hand-made sample files, and answers request.rebuild
with "accepted" then a new pipeline.setup_completed. No build, no LLM: canned messages from fixtures/.
"""
import json
from pathlib import Path
from typing import Any, Dict, List, Optional

from shared.messages import Message, MessageType

FIXTURES = Path(__file__).resolve().parent / "fixtures"
TOPIC = "pipeline:setup"


def _fixture(name: str) -> Dict[str, Any]:
    return json.loads((FIXTURES / name).read_text(encoding="utf-8"))


class SetupStub:
    """Fake Setup Agent: modes "default" (setup succeeds) and "failed" (setup gives up)."""

    def __init__(self, mode: str = "default"):
        self.mode = mode
        self.published_messages: List[Message] = []
        self._handled: set = set()

    def _publish(self, msg: Message) -> Message:
        self.published_messages.append(msg)
        return msg

    def emit_setup_completed(self, commit: Optional[str] = None) -> Message:
        payload = _fixture("setup_completed.json")
        if commit:
            payload["commit"] = commit
        return self._publish(Message(
            type=MessageType.FINDING, kind="pipeline.setup_completed", sender="setup", to=[],
            topic=TOPIC, level=1, confidence=1.0,
            summary=f"Stub: pipeline ready for {payload['repo']} (sample files), commit {payload['commit'][:7]}",
            evidence={"stages_passed": 9, "fix_attempts": 0}, payload=payload,
        ))

    def emit_setup_failed(self) -> Message:
        payload = _fixture("setup_failed.json")
        return self._publish(Message(
            type=MessageType.FINDING, kind="pipeline.setup_failed", sender="setup", to=["human"],
            topic=TOPIC, level=1, confidence=1.0,
            summary=f"Stub: setup failed at {payload['stage']} ({payload['error_class']}) after {payload['attempts']} fixes",
            evidence={"attempts": payload["attempts"]}, payload=payload,
        ))

    def emit(self) -> Message:
        return self.emit_setup_failed() if self.mode == "failed" else self.emit_setup_completed()

    def handle(self, msg: Message) -> List[Message]:
        """request.rebuild -> "accepted" answer, then the setup result. Idempotent; other kinds ignored."""
        if msg.kind != "request.rebuild" or msg.id in self._handled:
            return []
        self._handled.add(msg.id)
        answer = self._publish(Message(
            type=MessageType.ANSWER, kind="request.rebuild", sender="setup", to=[msg.sender],
            topic=TOPIC, reply_to=msg.id, incident_id=msg.incident_id, level=1,
            summary="Stub: rebuild accepted",
            payload={"status": "accepted", "changed_paths": msg.payload.get("changed_paths", [])},
        ))
        return [answer, self.emit()]


===== agents/trainer/agent.py =====
"""Trainer agent (roadmap chapter 10, owner P4).

Retrains only when it is worth it, and forwards a new model only if it is
better overall and not worse for any group of wines.

One step follows the roadmap 5.8 loop:
  1. Look        read the inbox (request.retrain, request.do_not_use,
                 request.expected_change) and the batches table
  2. Understand  triggers, clean window, drift, live MAE, expected gain (analysis.py)
  3. Decide      llm.decide() between refuse and retrain_in_dry_run, with rule_decision() as fallback
  4. Try safely  train_compare dry run: fit on the window, compare with the live model per group
  5. Do          second decision on the comparison: send_to_deployer or reject_candidate
  6. Check       the gate is re-applied in code: a candidate that fails it is never forwarded
  7. Remember    every decision goes to trainer_runs (memory.py, owner P5, replaces this later)

    python -m agents.trainer        # long-running, on board.db (see INTEGRATION.md)
"""

from __future__ import annotations

import json
import os
import uuid
from dataclasses import dataclass, field
from pathlib import Path

try:  # the real loop (owner P3) once merged
    from shared.base_agent import BaseAgent
except ImportError:  # stand-in until then
    from shared.standins import BaseAgent

from shared import llm
from shared.dry_run import dry_run

from agents.trainer import analysis

ROOT = Path(__file__).resolve().parents[2]
MODEL_NAME = "wine-quality"

MENU = [
    {"name": "refuse", "level": 1, "params_schema": ["expected_gain", "threshold"],

    {"name": "request_clean_window", "level": 1, "params_schema": ["start", "end", "min_rows"],
     "success_check": "Data Checker answers with batch ids"},
    {"name": "retrain_in_dry_run", "level": 2, "params_schema": ["batch_ids"],
     "success_check": "run finished, metrics and comparison logged"},
    {"name": "register_candidate", "level": 1, "params_schema": ["run_id"],
     "success_check": "new version with alias candidate and all reference files"},
    {"name": "send_to_deployer", "level": 1, "params_schema": ["run_id"],

]

}


def load_settings(overrides: dict | None = None) -> dict:
    """agents.yaml (trainer thresholds, mlflow, db_url) + overrides."""
    cfg = {**DEFAULTS, "tracking_uri": os.environ.get("MLFLOW_TRACKING_URI", "http://localhost:5000"),
           "db_path": "board.db", "run_id": os.environ.get("RUN_ID", "dev")}
    path = Path(os.environ.get("AGENTS_YAML", ROOT / "agents.yaml"))
    if path.exists():
        import yaml

        data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
        cfg.update(((data.get("agents") or {}).get("trainer") or {}).get("thresholds") or {})
        cfg["tracking_uri"] = (data.get("mlflow") or {}).get("tracking_uri", cfg["tracking_uri"])
        db_url = data.get("db_url") or ""
        if db_url.startswith("sqlite:///"):
            cfg["db_path"] = db_url[len("sqlite:///"):]
        cfg["run_id"] = data.get("run_id", cfg["run_id"])
    cfg.update(overrides or {})
    return cfg


@dataclass
class Pending:
    """A retrain the Trainer owes someone, and the candidate it produced."""

    incident_id: str
    reason: str
    requests: list[dict] = field(default_factory=list)  # messages to answer when done
    candidate: dict | None = None                        # train_compare outputs
    window: dict | None = None
    expected_gain: float | None = None


class TrainerAgent(BaseAgent):
    name = "trainer"
    menu = MENU

    def __init__(self, board, *, store: TrainerStore | None = None, settings: dict | None = None,
                 llm_client=None, llm_config: dict | None = None, now=None):
        super().__init__(board)
        self.cfg = load_settings(settings)
        self.store = store or TrainerStore(self.cfg["db_path"])
        self.llm_client = llm_client
        self.llm_config = llm_config
        self._now = now  # callable for tests; shared.clock otherwise
        self.retrain_requests: list[dict] = []
        self.do_not_use: set[str] = set()
        self.expected_change: dict | None = None  # {"change_id", "after_batch", "message"}
        self.pending: Pending | None = None
        self.log: list[dict] = []  # decisions taken, for tests and the README

    # ---- helpers --------------------------------------------------------
    def now(self) -> str:
        return self._now() if self._now else now_iso()

    def post(self, **fields) -> str:
        fields.setdefault("run_id", self.cfg["run_id"])
        return self.board.post({"sender": self.name, **fields})

    def _mlflow(self):
        import mlflow

        mlflow.set_tracking_uri(self.cfg["tracking_uri"])
        return mlflow

    def champion(self) -> dict:
        """Live model version, its reference files and its reference MAE."""
        mlflow = self._mlflow()
        mv = mlflow.MlflowClient().get_model_version_by_alias(MODEL_NAME, "champion")
        ref_dir = Path(mlflow.artifacts.download_artifacts(run_id=mv.run_id, artifact_path="reference"))
        read = lambda name: json.loads((ref_dir / name).read_text())  # noqa: E731
        return {
            "version": str(mv.version),
            "uri": f"models:/{MODEL_NAME}@champion",
            "psi_bins": read("psi_bins.json"),
            "importance": read("feature_importance.json"),
            "mae_ref": read("group_metrics.json")["overall"]["mae"],
        }

    # ---- step 1: look ---------------------------------------------------
    def handle(self, msg: dict) -> None:
        if msg["type"] == "answer":
            return
        kind = msg["kind"]
        if kind == "request.retrain":
            self.retrain_requests.append(msg)
        elif kind == "request.do_not_use":
            self.do_not_use.update(msg["payload"].get("batch_ids", []))
            self.board.answer(msg, sender=self.name, level=1, summary="Batches excluded from training windows",
                              payload={"decision": "done", "detail": f"excluded {msg['payload'].get('batch_ids')}"})
        elif kind == "request.expected_change":
            batches = self.store.batches(self.now())
            after = batches["batch_id"].iloc[-1] if len(batches) else None
            self.expected_change = {"change_id": msg["payload"].get("change_id"), "after_batch": after, "message": msg}
            self.board.answer(msg, sender=self.name, level=1,
                              summary="Will retrain on data after the change only",
                              payload={"decision": "accepted", "detail": f"window starts after {after}",
                                       "task": "retrain_on_post_change_data", "done_at": None})
        elif kind == "test.ping":
            self.board.answer(msg, sender=self.name, level=0, summary="trainer: pong", payload={"ok": True})
        else:
            print(f"[trainer] ignoring {kind} {msg['id']}")

    def observe(self) -> dict:
        return {"now": self.now(), "batches": self.store.batches(self.now()), "last": self.store.last_training()}

    # ---- step 2: understand --------------------------------------------
    def analyze(self, obs: dict) -> dict:
        batches = obs["batches"]
        clean = batches[batches["status"].isin(["passed", "repaired"]) & ~batches["batch_id"].isin(self.do_not_use)]
        last_batch = (obs["last"] or {}).get("last_batch")
        new_clean = clean[clean["batch_id"] > last_batch] if last_batch else clean
        due = len(new_clean) >= self.cfg["retrain_every_batches"]
        requested = bool(self.retrain_requests)
        after = self.expected_change["after_batch"] if self.expected_change else None
        if not (requested or due):
            return {"flag": False}
        retry = '"dry_run_failed"' in ((obs["last"] or {}).get("detail") or "")  # a failed run may work now: try again
        if requested and obs["last"] and last_batch and new_clean.empty and not self.expected_change and not retry:
            return {"flag": True, "already_done": obs["last"]}

        window = analysis.choose_window(batches, size=self.cfg["window_batches"], eval_batches=self.cfg["eval_batches"],
                                        after=after, do_not_use=self.do_not_use)
        if not window.eval_ids:  # nothing clean and labelled yet: no need to load the live model
            return {"flag": True, "nodes": [f"model:{MODEL_NAME}"], "trigger": "request" if requested else "schedule",
                    "window": window, "no_data": True, "numbers": {
                        "champion_version": None, "mae_ref": None, "mae_now": None, "mae_rise_pct": 0.0,
                        "mae_rise_pct_low95": 0.0, "weighted_psi": 0.0, "top_psi": {}, "importance_of_top_psi": {},
                        "new_rows": 0, "train_rows": 0, "expected_gain": 0.0, "threshold": self.cfg["min_gain_pct"],
                        "train_batches": 0, "eval_batches": 0, "excluded": window.excluded,
                        "expected_change": self.expected_change["change_id"] if self.expected_change else None}}
        champ = self.champion()
        recent_ids = (window.train_ids + window.eval_ids)[-self.cfg["drift_batches"]:]
        recent = self.store.rows(recent_ids, obs["now"])
        drift = analysis.drift_table(recent, champ["psi_bins"], champ["importance"]) if len(recent) else {"psi": {}, "weighted_psi": 0.0}
        # live-model error is measured on the newest rise_batches clean batches (eval batches included):
        # 500 rows leave a +-9% bootstrap interval on the MAE rise, too wide to see scenario D
        rise_ids = (window.train_ids + window.eval_ids)[-self.cfg["rise_batches"]:]
        eval_rows = self.store.rows(rise_ids, obs["now"])
        rise = {"mae_now": None, "rise_pct": 0.0, "rise_pct_low": 0.0}
        if len(eval_rows):
            from baseline import train

            X, y = train.prepare_features_train(eval_rows)
            errors = (self._mlflow().pyfunc.load_model(champ["uri"]).predict(X) - y).abs().to_numpy()
            rise = analysis.mae_rise(errors, champ["mae_ref"])
        mae_now, mae_rise_pct = rise["mae_now"], rise["rise_pct"]
        train_rows = len(self.store.rows(window.train_ids, obs["now"])) if window.train_ids else 0
        labelled_new = new_clean[new_clean["labelled"]]["batch_id"].tolist()
        new_rows = len(self.store.rows(labelled_new, obs["now"])) if labelled_new else 0
        gain = analysis.estimate_gain(drift["weighted_psi"], rise["rise_pct_low"], new_rows, self.cfg["gain_weights"])
        top_psi = dict(sorted(drift["psi"].items(), key=lambda kv: -kv[1])[:3])
        return {
            "flag": True,
            "nodes": [f"model:{MODEL_NAME}"],
            "trigger": "request" if requested else "schedule",
            "window": window,
            "numbers": {
                "champion_version": champ["version"],
                "mae_ref": round(champ["mae_ref"], 4),
                "mae_now": round(mae_now, 4) if mae_now else None,
                "mae_rise_pct": round(mae_rise_pct, 2),
                "mae_rise_pct_low95": round(rise["rise_pct_low"], 2),
                "weighted_psi": round(drift["weighted_psi"], 4),
                "top_psi": {k: round(v, 3) for k, v in top_psi.items()},
                "importance_of_top_psi": {k: round(champ["importance"].get(k, 0.0), 4) for k in top_psi},
                "new_rows": new_rows,
                "train_rows": train_rows,
                "expected_gain": round(gain, 2),
                "threshold": self.cfg["min_gain_pct"],
                "train_batches": len(window.train_ids),
                "eval_batches": len(window.eval_ids),
                "excluded": window.excluded,
                "expected_change": self.expected_change["change_id"] if self.expected_change else None,
            },
        }

    # ---- step 3: decide -------------------------------------------------
    def context(self, analysis_: dict, inbox: list) -> dict:
        past = []
        last = self.store.last_training()
        if last:
            past.append({k: last[k] for k in ("decision", "version", "expected_gain", "overall_gain_pct")})
        return {"numbers": analysis_["numbers"], "trigger": analysis_["trigger"],
                "requests": [m["summary"] for m in self.retrain_requests], "past_retrains": past}

    def rule_decision(self, analysis_: dict) -> dict:
        n = analysis_["numbers"]
        if analysis_.get("no_data"):
            return {"action": "refuse", "level": 1, "params": {"expected_gain": 0.0, "threshold": n["threshold"]},
                    "reason": "no clean labelled batches to train or evaluate on yet", "confidence": 0.95}
        if n["expected_gain"] < n["threshold"]:
            return {"action": "refuse", "level": 1,
                    "params": {"expected_gain": n["expected_gain"], "threshold": n["threshold"]},
                    "reason": f"retraining would lower the error (MAE) by only {n['expected_gain']}%, less than the {n['threshold']}% needed",
                    "confidence": 0.8}
        if n["train_rows"] < self.cfg["min_train_rows"]:
            return {"action": "refuse", "level": 1,
                    "params": {"expected_gain": n["expected_gain"], "threshold": n["threshold"]},
                    "reason": f"only {n['train_rows']} clean labelled rows to train on", "confidence": 0.8}
        return {"action": "retrain_in_dry_run", "level": 2,
                "params": {"batch_ids": analysis_["window"].train_ids},
                "reason": f"live error (MAE) went up {n['mae_rise_pct']}%; retraining should lower it by {n['expected_gain']}% (need {n['threshold']}%)",
                "confidence": 0.7}

    def decide(self, ctx: dict, menu: list, fallback: dict, incident_id: str) -> llm.Decision:
        return llm.decide(self, ctx, menu, fallback, incident_id=incident_id,
                          client=self.llm_client, config=self.llm_config)

    # ---- the step -------------------------------------------------------
    def step(self) -> None:
        for msg in self.board.read(self.name):
            self.safe_handle(msg)
        self.board.heartbeat(self.name)
        if self.pending and self.pending.candidate:
            return self.finish_candidate()
        try:
            obs = self.observe()
            for msg in self.board.read(self.name):  # a request posted with the batches just seen (same transaction)
                self.safe_handle(msg)
            result = self.analyze(obs)
        except Exception as exc:  # MLflow or board.db unreachable: say so, try again next step
            if self.retrain_requests:
                self._answer_requests("failed", f"could not analyse: {type(exc).__name__}: {exc}"[:200])
            self.log.append({"phase": "analyze", "error": repr(exc)})
            return None
        if not result["flag"]:
            return None
        if result.get("already_done"):  # asked again, but nothing new since the last training
            last = result["already_done"]
            done = last["decision"] == "forwarded"
            detail = (f"candidate v{last['version']} was already sent to the Deployer" if done else
                      f"already decided '{last['decision']}' on these batches; no new clean batches since {last['last_batch']}")
            self.log.append({"phase": "retrain", "action": "already_done", "source": "rules", "reason": detail})
            return self._answer_requests("done" if done else "refused", detail,
                                         {"version": last.get("version")} if done else None)
        incident_id = next((m["incident_id"] for m in self.retrain_requests if m.get("incident_id")), None) \
            or "inc-" + uuid.uuid4().hex[:8]
        rule = self.rule_decision(result)
        if result.get("no_data"):  # nothing to weigh up: no LLM call
            decision = llm.Decision(**rule, source="rules")
        else:
            decision = self.decide(self.context(result, []), RETRAIN_MENU, rule, incident_id)
        self.log.append({"phase": "retrain", **decision.model_dump()})
        if decision.action == "retrain_in_dry_run":
            return self.retrain(result, decision, incident_id)
        return self.refuse(result, decision, incident_id,
                           reason_code="not_enough_data" if result.get("no_data") else "gain_below_threshold")

    # ---- step 4-5: act --------------------------------------------------
    def _answer_requests(self, decision: str, detail: str, extra: dict | None = None) -> None:
        for req in self.retrain_requests:
            self.board.answer(req, sender=self.name, level=1, summary=f"Retrain request {decision}: {detail}"[:160],
                              payload={"decision": decision, "detail": detail, **(extra or {})})
        self.retrain_requests = []

    def refuse(self, result: dict, decision: llm.Decision, incident_id: str, reason_code: str = "gain_below_threshold") -> None:
        n = result["numbers"]
        window = result["window"].as_payload()
        self.post(type="finding", kind="training.decision", to=["monitor"], topic=f"model:{MODEL_NAME}",
                  incident_id=incident_id, level=1, confidence=decision.confidence,
                  summary=f"Retrain refused: {decision.reason}"[:160],
                  evidence={k: n[k] for k in ("weighted_psi", "mae_rise_pct", "mae_rise_pct_low95", "new_rows", "top_psi", "importance_of_top_psi")},
                  payload={"decision": "refuse", "expected_gain": n["expected_gain"], "threshold": n["threshold"],
                           "data_window": window, "reason_code": reason_code, "decided_by": decision.source})
        self._answer_requests("refused", decision.reason)
        last_batch = window["batch_ids"][-1] if window["batch_ids"] else None
        self.store.record(incident_id=incident_id, decision="refuse", expected_gain=n["expected_gain"],
                          train_ids=window["batch_ids"], eval_ids=window["eval_batch_ids"],
                          last_batch=(window["eval_batch_ids"] or [last_batch])[-1],
                          detail={"reason": decision.reason, "reason_code": reason_code})

    def retrain(self, result: dict, decision: llm.Decision, incident_id: str) -> None:
        n, window = result["numbers"], result["window"]
        train_ids = [b for b in decision.params.get("batch_ids", window.train_ids) if b in window.train_ids] or window.train_ids
        res = dry_run("train_compare", {
            "db_path": str(Path(self.store.db_path).resolve()), "train_ids": train_ids, "eval_ids": window.eval_ids,
            "now": self.now(), "tracking_uri": self.cfg["tracking_uri"],
            "live_model_uri": f"models:/{MODEL_NAME}@champion", "min_rows": self.cfg["min_train_rows"],
            "min_gain_pct": self.cfg["min_gain_pct"], "max_group_worse_pct": self.cfg["max_group_worse_pct"],
            "resamples": self.cfg["resamples"],
        }, timeout_s=300)
        if not res.ok:  # dry run failed: fall back to the safest action
            lines = [ln.strip() for ln in res.log_tail.splitlines() if ln.strip() and not ln.startswith("exit code")]
            safe = llm.Decision(action="refuse", level=1, params={}, source="rules",
                                reason=f"training dry run failed: {(lines or ['no output'])[-1][-150:]}")
            return self.refuse(result, safe, incident_id, reason_code="dry_run_failed")
        self.post(type="finding", kind="training.decision", to=["monitor"], topic=f"model:{MODEL_NAME}",
                  incident_id=incident_id, level=2, confidence=decision.confidence,
                  summary=f"Retrained on {len(train_ids)} clean batches: {decision.reason}"[:160],
                  evidence={k: n[k] for k in ("weighted_psi", "mae_rise_pct", "mae_rise_pct_low95", "new_rows")},
                  payload={"decision": "retrain", "expected_gain": n["expected_gain"], "threshold": n["threshold"],
                           "data_window": {**window.as_payload(), "batch_ids": train_ids},
                           "reason_code": "requested" if result["trigger"] == "request" else "schedule",
                           "decided_by": decision.source, "candidate_run_id": res.outputs["run_id"]})
        self.pending = Pending(incident_id=incident_id, reason=decision.reason, requests=self.retrain_requests,
                               candidate=res.outputs, window={"train_ids": train_ids, "eval_ids": window.eval_ids},
                               expected_gain=n["expected_gain"])
        self.retrain_requests = []
        self.finish_candidate()

    def finish_candidate(self) -> None:
        p = self.pending
        cand, verdict = p.candidate, p.candidate["gate"]
        report = cand["report"]
        ctx = {"gate": verdict, "overall": report["overall"], "expected_gain": p.expected_gain,
               "min_gain_pct": self.cfg["min_gain_pct"], "max_group_worse_pct": self.cfg["max_group_worse_pct"]}
        rule = ({"action": "send_to_deployer", "level": 1, "params": {"run_id": cand["run_id"]},
                 "reason": f"error (MAE) {verdict['overall_gain_pct']:.1f}% lower, no alcohol or acidity group has a higher error", "confidence": 0.85}
                if verdict["passed"] else
                {"action": "reject_candidate", "level": 1, "params": {"run_id": cand["run_id"]},
                 "reason": self._reject_reason(verdict), "confidence": 0.9})
        decision = self.decide(ctx, FORWARD_MENU, rule, p.incident_id)
        if decision.action == "send_to_deployer" and not verdict["passed"]:
            # 6. Check: the gate is code, not LLM judgement
            decision = llm.Decision(**{**rule, "source": "rules"})
        self.log.append({"phase": "forward", **decision.model_dump()})
        self.retrain_requests = p.requests
        if decision.action == "send_to_deployer":
            self.send_to_deployer(decision)
        else:
            self.reject(decision)
        self.pending = None

    def _reject_reason(self, verdict: dict) -> str:
        g = verdict["overall_gain_pct"]
        overall = f"error (MAE) {g:.1f}% lower overall" if g >= 0 else f"error (MAE) {-g:.1f}% higher overall"
        if verdict["failing_groups"]:
            groups = ", ".join(f"{f['group']} {f['band']}" for f in verdict["failing_groups"])
            return f"{overall} and higher for {groups}" if g < 0 else f"{overall} but higher for {groups}"
        if g < 0:
            return overall
        return f"{overall}, less than the {self.cfg['min_gain_pct']}% needed"

    def send_to_deployer(self, decision: llm.Decision) -> None:
        from baseline import train

        p, cand = self.pending, self.pending.candidate
        report = cand["report"]
        champ_version = self.champion()["version"]
        version = train.register_run(cand["run_id"], alias="candidate", tracking_uri=self.cfg["tracking_uri"],
                                     tags={"incident_id": p.incident_id, "train_batches": ",".join(p.window["train_ids"])})
        by_group = {g: {b: round(s["delta_pct"], 2) for b, s in bands.items() if s.get("counts")}
                    for g, bands in report["groups"].items()}
        self.post(type="request", kind="request.deploy_candidate", to=["deployer"], topic=f"model:{MODEL_NAME}@{version}",
                  incident_id=p.incident_id, level=2, confidence=decision.confidence,
                  summary=(f"Candidate v{version} passes the gate: error (MAE) {report['overall']['mae_new']:.3f} vs "
                           f"{report['overall']['mae_live']:.3f} live, {-report['overall']['delta_pct']:.1f}% lower")[:160],
                  evidence={"eval_rows": report["overall"]["n"], "bootstrap_resamples": report["resamples"]},
                  expires_at=None,
                  payload={"model_uri": f"models:/{MODEL_NAME}@candidate", "version": version,
                           "champion_version": champ_version,
                           "metrics": {"overall": {"mae_candidate": round(report["overall"]["mae_new"], 4),
                                                   "mae_champion": round(report["overall"]["mae_live"], 4),
                                                   "delta_pct": round(report["overall"]["delta_pct"], 2)},
                                       "by_group": by_group},
                           "comparison_ref": cand["comparison_ref"]})
        self._answer_requests("done", f"candidate v{version} sent to the Deployer")
        self._finish_expected_change(version)
        self.store.record(run_id=cand["run_id"], incident_id=p.incident_id, decision="forwarded", version=version,
                          train_ids=p.window["train_ids"], eval_ids=p.window["eval_ids"],
                          last_batch=p.window["eval_ids"][-1], expected_gain=p.expected_gain,
                          overall_gain_pct=cand["gate"]["overall_gain_pct"], detail={"reason": decision.reason})

    def reject(self, decision: llm.Decision) -> None:
        p, cand = self.pending, self.pending.candidate
        verdict = cand["gate"]
        self.post(type="finding", kind="model.candidate_rejected", to=["monitor", "human"],
                  topic=f"model:{MODEL_NAME}", incident_id=p.incident_id, level=1, confidence=decision.confidence,
                  summary=f"Candidate rejected: {decision.reason}"[:160],
                  evidence={"mae_live": cand["report"]["overall"]["mae_live"], "mae_candidate": cand["report"]["overall"]["mae_new"],
                            "bootstrap_resamples": cand["report"]["resamples"]},
                  payload={"version": f"run:{cand['run_id']}", "overall_delta": round(-verdict["overall_gain_pct"] / 100, 4),
                           "failing_groups": verdict["failing_groups"], "comparison_ref": cand["comparison_ref"]})
        self._answer_requests("failed", decision.reason)
        self.store.record(run_id=cand["run_id"], incident_id=p.incident_id, decision="rejected",
                          train_ids=p.window["train_ids"], eval_ids=p.window["eval_ids"],
                          last_batch=p.window["eval_ids"][-1], expected_gain=p.expected_gain,
                          overall_gain_pct=verdict["overall_gain_pct"], detail={"reason": decision.reason})

    def _finish_expected_change(self, version: str) -> None:
        if not self.expected_change:
            return
        msg = self.expected_change["message"]
        self.board.answer(msg, sender=self.name, level=1, summary=f"Retrained on post-change data; candidate v{version} sent",
                          payload={"decision": "done", "detail": f"candidate v{version}",
                                   "task": "retrain_on_post_change_data", "done_at": self.now()})
        self.expected_change = None


===== agents/trainer/analysis.py =====
"""Trainer statistics: no LLM here (roadmap 10.1, 10.4).

- psi / weighted_psi: input drift of a data window against the reference bins
- estimate_gain: v0 gain estimate that gates retraining
- choose_window: which clean batches to train and evaluate on
- compare / gate: overall and per-group MAE of a candidate vs the live model,
  with a paired bootstrap (500 resamples)
"""

from __future__ import annotations

import math
from dataclasses import dataclass, field

import numpy as np
import pandas as pd

GROUPS = ("alcohol_band", "acidity_band", "quality_band")
# The gate vetoes only on groups made from the inputs. quality_band is made from the label itself:
# when every score drops one point, wines move between quality bands, and the old model (which predicts
# near the mean) looks good on band 6 by chance. That group is reported, never a reason to reject.
VETO_GROUPS = ("alcohol_band", "acidity_band")
EPS = 1e-4


# --------------------------------------------------------------------------
# Drift
# --------------------------------------------------------------------------
def psi(values: np.ndarray, cuts: list[float], ref_share: list[float]) -> float:
    """Population stability index of values against reference bins (cut points + shares)."""
    values = np.asarray(values, dtype=float)
    values = values[~np.isnan(values)]
    if len(values) == 0:
        return float("nan")
    idx = np.searchsorted(np.asarray(cuts), values, side="left")
    cur = np.bincount(idx, minlength=len(cuts) + 1) / len(values)
    ref = np.clip(np.asarray(ref_share), EPS, None)
    cur = np.clip(cur, EPS, None)
    return float(np.sum((cur - ref) * np.log(cur / ref)))


def drift_table(frame: pd.DataFrame, psi_bins: dict, importance: dict) -> dict:
    """PSI per feature and the importance-weighted PSI (sum(imp * psi) / sum(imp))."""
    per_feature = {}
    for feature, bins in psi_bins["features"].items():
        if feature in frame:
            per_feature[feature] = psi(frame[feature].to_numpy(), bins["cuts"], bins["ref_share"])
    total_imp = sum(importance.get(f, 0.0) for f in per_feature) or 1.0
    weighted = sum(importance.get(f, 0.0) * v for f, v in per_feature.items() if not math.isnan(v)) / total_imp
    return {"psi": per_feature, "weighted_psi": float(weighted)}


# --------------------------------------------------------------------------
# Gain estimate v0 (roadmap 10.4 W2 Wed): gain = a*weighted_psi + b*mae_rise + c*log(new_rows)
# --------------------------------------------------------------------------
DEFAULT_GAIN_WEIGHTS = {"a": 10.0, "b": 0.5, "c": 0.3}


def estimate_gain(weighted_psi: float, mae_rise_pct: float, new_rows: int, weights: dict | None = None) -> float:
    """Expected MAE improvement of a retrain, in percent.

    weighted_psi in PSI units, mae_rise_pct = how much worse the live model is
    than at training time (percent, negative clipped to 0), new_rows = clean
    labelled rows the live model has not seen. Hand-set weights, to be tuned.
    """
    w = {**DEFAULT_GAIN_WEIGHTS, **(weights or {})}
    return float(
        w["a"] * max(weighted_psi, 0.0)
        + w["b"] * max(mae_rise_pct, 0.0)
        + w["c"] * math.log1p(max(new_rows, 0) / 1000)
    )


def mae_rise(abs_errors: np.ndarray, mae_ref: float, *, resamples: int = 500, seed: int = 0) -> dict:
    """How much worse the live model is now than at training time, in percent.

    Returns the point estimate and the lower 95% bootstrap bound. The gain
    estimate uses the lower bound: on 500 rows the point estimate moves by
    about +-5% from sampling alone, which would otherwise trigger retrains on noise.
    """
    abs_errors = np.asarray(abs_errors, dtype=float)
    rng = np.random.default_rng(seed)
    boot = abs_errors[rng.integers(0, len(abs_errors), size=(resamples, len(abs_errors)))].mean(axis=1)
    to_pct = lambda m: float(100 * (m / mae_ref - 1))  # noqa: E731
    return {"mae_now": float(abs_errors.mean()), "rise_pct": to_pct(abs_errors.mean()),
            "rise_pct_low": to_pct(np.percentile(boot, 2.5))}


# --------------------------------------------------------------------------
# Window choice
# --------------------------------------------------------------------------
@dataclass
class Window:
    train_ids: list[str]
    eval_ids: list[str]
    excluded: dict[str, str] = field(default_factory=dict)  # batch_id -> reason

    def as_payload(self) -> dict:
        ids = self.train_ids + self.eval_ids
        return {
            "start": ids[0] if ids else None,
            "end": ids[-1] if ids else None,
            "batch_ids": self.train_ids,
            "eval_batch_ids": self.eval_ids,
            "excluded": self.excluded,
        }


def choose_window(
    batches: pd.DataFrame,
    *,
    size: int | None = None,
    eval_batches: int = 3,
    after: str | None = None,
    do_not_use: set[str] | None = None,
) -> Window:
    """Pick training and evaluation batches.

    batches: rows of the batches table (batch_id, status, labelled) in time order.
    Only passed / repaired batches count; do_not_use batches and anything up to
    ``after`` (the last batch before an expected change) are left out. The
    newest ``eval_batches`` labelled clean batches are held out for evaluation;
    the ``size`` batches before them (or all) are the training window.
    """
    excluded: dict[str, str] = {}
    keep = []
    seen_after = after is None
    for row in batches.itertuples(index=False):
        if not seen_after:
            excluded[row.batch_id] = "before_expected_change"
            seen_after = row.batch_id == after
            continue
        if row.status not in ("passed", "repaired"):
            excluded[row.batch_id] = row.status
        elif do_not_use and row.batch_id in do_not_use:
            excluded[row.batch_id] = "do_not_use"
        elif not row.labelled:
            excluded[row.batch_id] = "no_labels_yet"
        else:
            keep.append(row.batch_id)
    eval_ids = keep[-eval_batches:] if eval_batches else []
    train_ids = keep[: len(keep) - len(eval_ids)]
    if size:
        train_ids = train_ids[-size:]
    return Window(train_ids, eval_ids, excluded)


# --------------------------------------------------------------------------
# Comparison and gate
# --------------------------------------------------------------------------
def _paired_bootstrap(err_new: np.ndarray, err_live: np.ndarray, rng: np.random.Generator, n: int) -> tuple[float, float]:
    idx = rng.integers(0, len(err_new), size=(n, len(err_new)))
    diffs = err_new[idx].mean(axis=1) - err_live[idx].mean(axis=1)
    lo, hi = np.percentile(diffs, [2.5, 97.5])
    return float(lo), float(hi)


def compare(
    frame: pd.DataFrame,
    *,
    y: str = "quality",
    new: str = "pred_new",
    live: str = "pred_live",
    resamples: int = 500,
    min_group: int = 50,
    seed: int = 0,
) -> dict:
    """Overall and per-group MAE of new vs live on the same evaluation rows."""
    rng = np.random.default_rng(seed)
    err_new = (frame[y] - frame[new]).abs().to_numpy()
    err_live = (frame[y] - frame[live]).abs().to_numpy()

    def block(mask: np.ndarray) -> dict:
        en, el = err_new[mask], err_live[mask]
        lo, hi = _paired_bootstrap(en, el, rng, resamples)
        return {
            "n": int(mask.sum()),
            "mae_new": float(en.mean()),
            "mae_live": float(el.mean()),
            "delta": float(en.mean() - el.mean()),
            "delta_pct": float(100 * (en.mean() - el.mean()) / el.mean()) if el.mean() else 0.0,
            "delta_ci95": [lo, hi],
        }

    report = {"overall": block(np.ones(len(frame), dtype=bool)), "groups": {}, "resamples": resamples}
    for group in GROUPS:
        if group not in frame:
            continue
        report["groups"][group] = {}
        for band in sorted(frame[group].dropna().unique()):
            mask = (frame[group] == band).to_numpy()
            if mask.sum() < min_group:
                report["groups"][group][str(band)] = {"n": int(mask.sum()), "counts": False}
                continue
            report["groups"][group][str(band)] = {**block(mask), "counts": True}
    return report


def gate(report: dict, *, min_gain_pct: float = 2.0, max_group_worse_pct: float = 5.0) -> dict:
    """Pass when overall MAE improves by >= min_gain_pct and no counted group is clearly worse.

    A group fails when it is worse by more than max_group_worse_pct of its live
    MAE (point estimate) AND the paired bootstrap says the difference is real
    (lower 95% bound above 0). Only VETO_GROUPS can fail; a worse quality_band
    band is listed in ``label_groups_worse`` for the record.

    Calibration note: roadmap 10.1 starts from "upper 95% bound below +5%".
    Measured on 3 seeds x 3 window sizes, that rule flagged a "worse" group
    in 8 of 9 candidates trained on unchanged data (noise alone crosses +5%), and rejected
    the scenario D candidate for being 2-3% worse on low-alcohol wines while
    29% better on high-alcohol ones. This rule keeps F rejected (+82% on
    alcohol > 11.5) and lets D through.
    """
    overall = report["overall"]
    gain_pct = -overall["delta_pct"]
    failing, label_worse = [], []
    for group, bands in report["groups"].items():
        for band, stats in bands.items():
            if not stats.get("counts"):
                continue
            limit = max_group_worse_pct / 100 * stats["mae_live"]
            if stats["delta"] > limit and stats["delta_ci95"][0] > 0:
                (failing if group in VETO_GROUPS else label_worse).append({
                    "group": group, "band": band, "n": stats["n"],
                    "mae_live": round(stats["mae_live"], 4), "mae_candidate": round(stats["mae_new"], 4),
                    "delta_pct": round(stats["delta_pct"], 2),
                    "delta_ci95": [round(v, 4) for v in stats["delta_ci95"]], "limit": round(limit, 4),
                })
    return {
        "passed": gain_pct >= min_gain_pct and not failing,
        "overall_gain_pct": round(gain_pct, 3),
        "overall_ok": gain_pct >= min_gain_pct,
        "failing_groups": failing,
        "label_groups_worse": label_worse,
    }


===== agents/trainer/prompt.md =====
You are the Trainer agent of a wine-quality MLOps pipeline. You decide whether
retraining the wine-quality model (ElasticNet on 11 lab measurements) is worth
it, which clean data window to train on, and whether a new candidate is good
enough to send to the Deployer.

The model is scored by MAE (mean absolute error, in quality points). LOWER MAE IS BETTER.
"expected_gain" in the context is the expected REDUCTION of MAE, in percent; "mae_rise_pct" is how
much the live model's MAE went UP (got worse).

Rules:
- Retrain only when the expected MAE reduction (expected_gain) is at or above the threshold in the context.
- Never train on quarantined batches, or on data from before an active expected change.
- Forward a candidate only if its MAE is lower overall by at least the threshold AND no wine
  group's MAE is higher beyond its limit (alcohol and acidity groups; quality bands are only reported). If any group fails, reject it and list the failing groups.
- Quote the numbers you used in "reason".

"""Fake Trainer (roadmap 5.1 rule 5 and 10.6) so the other five agents can test alone.

- answers request.retrain with accepted, then posts a training.decision
  (retrain or refuse, chosen by the fixture set);
- after a retrain, posts request.deploy_candidate for a hand-registered

- answers request.expected_change with done.

Fixture set: STUB_FIXTURE=retrain (default, scenario D) | refuse (E) | reject (F).

    python -m agents.trainer.stub            # needs the real board (shared/board.py)

from __future__ import annotations

import copy
import json
import os
import time
from pathlib import Path

FIXTURES = Path(__file__).resolve().parent / "fixtures"
FIXTURE_SETS = {
    "retrain": ["training_decision_retrain.json", "deploy_candidate.json"],
    "refuse": ["training_decision_refuse.json"],
    "reject": ["training_decision_retrain.json", "candidate_rejected.json"],
}
ENVELOPE_KEYS = ("id", "time", "status", "reply_to")


def load_fixture(name: str) -> dict:
    msg = json.loads((FIXTURES / name).read_text())
    for key in ENVELOPE_KEYS:  # the board fills these in
        msg.pop(key, None)
    return msg


class TrainerStub:
    name = "trainer"

    def __init__(self, board, fixture_set: str | None = None):
        self.board = board
        self.fixture_set = fixture_set or os.environ.get("STUB_FIXTURE", "retrain")
        if self.fixture_set not in FIXTURE_SETS:
            raise ValueError(f"unknown fixture set {self.fixture_set!r}; use one of {sorted(FIXTURE_SETS)}")
        self.handled: set[str] = set()

    def handle(self, msg: dict) -> None:
        if msg["id"] in self.handled or msg["type"] == "answer":
            return
        self.handled.add(msg["id"])
        if msg["kind"] == "request.retrain":
            self.board.answer(msg, sender=self.name, level=1, summary="Retrain request accepted (stub)",
                              payload={"decision": "accepted", "detail": f"stub fixture set {self.fixture_set}"})
            for name in FIXTURE_SETS[self.fixture_set]:
                out = copy.deepcopy(load_fixture(name))
                out["incident_id"] = msg.get("incident_id")
                out["run_id"] = msg.get("run_id", "dev")
                self.board.post(out)
        elif msg["kind"] == "request.expected_change":
            self.board.answer(msg, sender=self.name, level=1, summary="Expected change handled (stub)",
                              payload={"decision": "done", "detail": "stub", "task": "retrain_on_post_change_data",
                                       "done_at": msg["time"]})

    def step(self) -> None:
        for msg in self.board.read(self.name):
            self.handle(msg)

    while True:

        time.sleep(2)

"""

    python -m harness.demo_checker          # decisions by the rules
    python -m harness.demo_checker --llm    # decisions by the LLM of agents.yaml (needs .env)

import contextlib
import io
import json

from pathlib import Path

from shared.tables import Tables

CASES = [

    ("data.valid_shift.white_wine", "un lot de vins BLANCS (valide, mais différent)"),

def run_case(fault_id: str, llm=None):
    from agents.checker.agent import CheckerAgent
    from agents.checker.settings import CheckerSettings

    clock.configure(mode="frozen", start=datetime(2026, 3, 14, tzinfo=timezone.utc))

    def do_GET(self):
        url = urlparse(self.path)
        if url.path in ("/", "/index.html"):
            return self.send(200, PAGE.read_bytes(), "text/html; charset=utf-8")
        route = ROUTES.get(url.path)

        try:
            with _lock:
                body = route(query)
            return self.send(200, json.dumps(body, default=str).encode(), "application/json")

        self.send_response(code)
        self.send_header("Content-Type", ctype)
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, fmt, *args):     # keep the console quiet during the demo
        pass

===== harness/evaluate_checker.py =====
"""
Measures the Data Checker against the targets of roadmap 9.6:
>= 90% of data faults decided correctly, <= 1 false alarm per 15 normal batches.

For each data fault: replay 10 clean batches, then the faulty batch 11, with the
real Checker (rules or LLM) and fake partners. A false alarm is a normal batch
that is not plainly passed (quarantined, repaired, or passed with a warning).

    python -m harness.evaluate_checker [--normal 45] [--seeds 3]
"""
from __future__ import annotations

import argparse
import contextlib
import io
import tempfile
from datetime import datetime, timezone
from pathlib import Path

from shared import clock
from shared.tables import Tables
from shared.testing import InMemoryBoard, SimpleIncidents, inprocess_dry_run
from harness.fault_runner import FaultRunner, all_faults
from harness.replay import ROOT, Replay

STATUS_OF = {"quarantine": "quarantined", "repair": "repaired",
             "pass": "passed", "pass_with_warning": "passed"}


def make_checker(work: Path, board, tables, llm=None):
    from agents.checker.agent import CheckerAgent
    from agents.checker.settings import CheckerSettings
    return CheckerAgent(board=board, incidents=SimpleIncidents(), tables=tables, llm=llm,
                        settings=CheckerSettings(data_dir=work / "checker"), dry_run=inprocess_dry_run)


def run(n_batches: int, fault=None, seed: int = 0, llm=None):
    clock.configure(mode="frozen", start=datetime(2026, 3, 14, tzinfo=timezone.utc))
    work = Path(tempfile.mkdtemp(prefix="eval_checker_"))
    board, tables = InMemoryBoard(), Tables(":memory:")
    runner = FaultRunner([fault], tables) if fault else None
    replay = Replay(tables, work / "batches", seed=seed, fault_runner=runner)
    checker = make_checker(work, board, tables, llm)
    with contextlib.redirect_stdout(io.StringIO()):
        for _ in range(n_batches):
            replay.next_batch()
            checker.step()
    return tables


def evaluate(normal: int = 45, seeds: int = 3, llm=None) -> dict:
    rows = []
    for fid, fault in all_faults().items():
        if fault.layer != "data":
            continue
        if "white_wine" in fault.inject and not (ROOT / "data" / "winequality-white.csv").exists():
            rows.append((fid, "skipped (needs data/winequality-white.csv)", "", None))
            continue
        for seed in range(seeds):
            tables = run(fault.after_batches + 1, fault, seed, llm)
            got = tables.batch(f"batch-{fault.after_batches + 1:04d}")
            wanted = {STATUS_OF[a] for a in fault.expected.get("acceptable_actions", [])}
            acceptable = fault.expected.get("acceptable_actions", [])
            if "pass" in acceptable:
                ok = got["status"] == "passed"
            elif "pass_with_warning" in acceptable:          # must pass AND warn
                ok = got["status"] == "passed" and got["reason"] != "pass"
            else:
                ok = got["status"] in wanted
            rows.append((fid, got["status"], got["reason"][:50], ok))

    false_alarms, warned = 0, 0
    for seed in range(seeds):
        tables = run(normal // seeds, None, 100 + seed, llm)
        for b in tables.batches():
            if b["status"] != "passed":
                false_alarms += 1
            elif b["reason"] != "pass":
                warned += 1
    graded = [r for r in rows if r[3] is not None]
    return {"faults": rows, "fault_accuracy": sum(r[3] for r in graded) / max(len(graded), 1),
            "normal_batches": (normal // seeds) * seeds, "false_alarms": false_alarms,
            "warnings_on_normal": warned}


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--normal", type=int, default=45)
    p.add_argument("--seeds", type=int, default=3)
    p.add_argument("--llm", action="store_true",
                   help="decide with the real LLM of agents.yaml (needs .env) instead of the rules")
    a = p.parse_args()
    llm, name, first_call = None, "", 0
    if a.llm:
        from harness.demo_checker import real_llm
        llm, name = real_llm()
        first_call = len(llm.CALL_LOG)
    r = evaluate(a.normal, a.seeds, llm)
    for fid, status, reason, ok in r["faults"]:
        print(f"{'ok ' if ok else ('-- ' if ok is None else 'NO ')} {fid:42s} {status:12s} {reason}")
    print(f"\nfaults decided correctly: {r['fault_accuracy']:.0%}   (target >= 90%)")
    per15 = 15 * (r["false_alarms"] + r["warnings_on_normal"]) / r["normal_batches"]
    print(f"normal batches: {r['normal_batches']}, wrongly blocked/repaired: {r['false_alarms']}, "
          f"passed with a warning: {r['warnings_on_normal']}  -> {per15:.1f} alarms per 15 (target <= 1)")
    if llm is not None:
        report_llm(llm, name, first_call)


def report_llm(llm, name: str, first_call: int = 0) -> None:
    """LLM cost and reliability, from shared/llm.py's call log (one record per attempt).
    The same records are in the llm_calls table of the board.db named by agents.yaml."""
    calls = [c for c in llm.CALL_LOG[first_call:] if c.agent == "checker"]
    by = {o: sum(c.outcome == o for c in calls) for o in ("ok", "invalid", "error", "fallback")}
    print(f"\nLLM {name}: {len(calls)} calls, {by['ok']} valid decisions, {by['invalid']} invalid "
          f"answers, {by['error']} errors, {by['fallback']} fell back to the rules")
    if calls:
        lat = [c.latency_ms / 1000 for c in calls]
        tokens = [(c.tokens_in or 0) + (c.tokens_out or 0) for c in calls]
        print(f"latency mean {sum(lat) / len(lat):.2f}s, max {max(lat):.2f}s; "
              f"tokens mean {sum(tokens) / len(tokens):.0f}, total {sum(tokens)}")
    print(f"call log: table llm_calls in {llm.load_config().get('db_url')}")

===== harness/fakes.py =====
"""
Scripted fake agents for solo scenario runs (roadmap 5.1 rule 5).

A fake posts the messages its scenario script gives it at a given batch, and
answers every request or question addressed to it with a canned answer
(default: accepted). Times in payloads may be written "{now}" or "{now-3h}".
"""
from __future__ import annotations

import re
from datetime import timedelta

from shared import clock
from shared.messages import Message, MessageType

DEFAULT_ANSWERS = {"request.": {"decision": "accepted", "detail": "fake agent"},
                   "query.": {"changed": False, "batch_ids": [], "likely_cause": "none"}}


def render(value):
    """'{now-3h}' -> ISO time 3 simulated hours ago; recursive on dicts and lists."""
    if isinstance(value, dict):
        return {k: render(v) for k, v in value.items()}
    if isinstance(value, list):
        return [render(v) for v in value]
    if isinstance(value, str):
        m = re.fullmatch(r"\{now(?:-(\d+)h)?\}", value)
        if m:
            t = clock.now() - timedelta(hours=int(m.group(1) or 0))
            return t.isoformat().replace("+00:00", "Z")
    return value


class FakeAgent:
    def __init__(self, name: str, board, incidents, script: list[dict] | None = None,
                 answers: dict | None = None):
        self.name = name
        self.board = board
        self.incidents = incidents
        self.script = script or []
        self.answers = answers or {}
        self.received: list[Message] = []

    def post_scripted(self, batch_index: int) -> None:
        for item in self.script:
            if item.get("at_batch") != batch_index:
                continue
            topic = item["topic"]
            incident = self.incidents.open_or_join([topic], 3600 * 6)
            self.board.post(Message(
                type=MessageType(item.get("type", "question")), kind=item["kind"], sender=self.name,
                to=item.get("to", []), topic=topic, incident_id=incident, level=item.get("level", 1),
                summary=item.get("summary", f"{self.name} fake: {item['kind']}")[:160],
                payload=render(item.get("payload", {})), evidence=item.get("evidence", {}),
                time=clock.iso(), expires_at=render("{now}")))

    def respond(self) -> None:
        for msg in self.board.read(self.name):
            self.received.append(msg)
            if msg.type in (MessageType.REQUEST, MessageType.QUESTION) and self.name in msg.to:
                payload = self.answers.get(msg.kind) or next(
                    (v for k, v in DEFAULT_ANSWERS.items() if msg.kind.startswith(k)), {})
                self.board.answer(msg, sender=self.name, payload=dict(payload),
                                  summary=f"{self.name} fake answers {msg.kind}")


===== harness/fault_runner.py =====
"""
Fault runner (owner P3, roadmap 13.4). Loads every harness/faults/<layer>/*.yaml,
applies the active faults to the replayed feed, and writes the ground truth to
the `faults` table, which agents never read.

    python -m harness.fault_runner list
"""
from __future__ import annotations

import importlib
import inspect
import sys
from dataclasses import dataclass, field
from pathlib import Path
from typing import Callable, Optional

import pandas as pd
import yaml

FAULT_DIR = Path(__file__).resolve().parent / "faults"


def load_callable(ref: str) -> Callable:
    module, func = ref.split(":")
    return getattr(importlib.import_module(module), func)


@dataclass
class Fault:
    id: str
    layer: str
    target: dict
    after_batches: int
    duration: int
    applies_to: list[str]
    inject: str
    revert: str
    expected: dict = field(default_factory=dict)
    author: str = ""

    @classmethod
    def from_yaml(cls, path: Path) -> "Fault":
        d = yaml.safe_load(path.read_text(encoding="utf-8"))
        return cls(id=d["id"], layer=d["layer"], target=d.get("target", {}),
                   after_batches=int(d["start"]["after_batches"]),
                   duration=int(d.get("duration", {}).get("batches", 1)),
                   applies_to=list(d.get("applies_to", ["batch"])), inject=d["inject"],
                   revert=d.get("revert", ""), expected=d.get("expected", {}),
                   author=d.get("author", ""))

    def active(self, batch_index: int) -> bool:
        """batch_index is 1-based: after_batches=10 hits batches 11 .. 10+duration."""
        return self.after_batches < batch_index <= self.after_batches + self.duration

    def apply(self, df: pd.DataFrame, batch_index: int) -> pd.DataFrame:
        """Two inject styles: ours f(df, **target), and the Trainer's f(batch, step, target)
        (harness/faults/model.py) where step counts the batches inside the fault from 0."""
        inject = load_callable(self.inject)
        if "step" in inspect.signature(inject).parameters:
            return inject(df, batch_index - self.after_batches - 1, dict(self.target))
        return inject(df, **self.target)


def all_faults() -> dict[str, Fault]:
    faults = {}
    for path in sorted(FAULT_DIR.glob("*/*.yaml")):
        f = Fault.from_yaml(path)
        faults[f.id] = f
    return faults


class FaultRunner:
    def __init__(self, faults: list[Fault], tables=None, run_id: str = "dev"):
        self.faults = faults
        self.tables = tables
        self.run_id = run_id
        self.recorded: set[str] = set()

    def apply(self, batch_index: int, df: pd.DataFrame, where: str,
              batch_id: Optional[str] = None) -> pd.DataFrame:
        for f in self.faults:
            if f.active(batch_index) and where in f.applies_to:
                df = f.apply(df, batch_index)
                if self.tables is not None and f.id not in self.recorded:
                    self.recorded.add(f.id)
                    self.tables.record_fault(f.id, f.layer, f.target, batch_id or str(batch_index),
                                             f.expected, self.run_id)
        return df


if __name__ == "__main__":
    if sys.argv[1:] == ["list"]:
        for fid, f in all_faults().items():
            print(f"{fid:45s} layer={f.layer:6s} batches {f.after_batches + 1}-{f.after_batches + f.duration}"
                  f"  expected={f.expected.get('acceptable_actions')}")

===== harness/faults/data.py =====
"""
Data-layer fault functions (author P3), referenced from harness/faults/data/*.yaml.

A fault is applied to the feed, so it reaches both the batch and the live
requests (roadmap 9.1). inject(df, **target) returns a broken copy; revert
functions exist for the file format but are no-ops, because every batch is a
fresh copy of the source data.
"""
from __future__ import annotations

from pathlib import Path

import numpy as np
import pandas as pd

ROOT = Path(__file__).resolve().parents[2]


def _rng(seed: int = 7) -> np.random.Generator:
    return np.random.default_rng(seed)


def rescale_column(df: pd.DataFrame, column: str, factor: float) -> pd.DataFrame:
    out = df.copy()
    out[column] = out[column] * factor
    return out


def null_values(df: pd.DataFrame, column: str, rate: float) -> pd.DataFrame:
    out = df.copy()
    out.loc[_rng().random(len(out)) < rate, column] = np.nan
    return out


def rename_column(df: pd.DataFrame, column: str, new_name: str) -> pd.DataFrame:
    return df.rename(columns={column: new_name})


def duplicate_rows(df: pd.DataFrame, rate: float) -> pd.DataFrame:
    extra = df.sample(frac=rate, random_state=3)
    return pd.concat([df, extra], ignore_index=True)


def add_column(df: pd.DataFrame, column: str, value=1.0) -> pd.DataFrame:
    out = df.copy()
    out[column] = value
    return out


def out_of_range(df: pd.DataFrame, column: str, rate: float, value: float) -> pd.DataFrame:
    out = df.copy()
    out.loc[_rng().random(len(out)) < rate, column] = value
    return out


def white_wine(df: pd.DataFrame, path: str = "data/winequality-white.csv") -> pd.DataFrame:
    """A valid batch from another population: should be a warning, not a quarantine."""
    source = ROOT / path
    if not source.exists():
        raise FileNotFoundError(f"{source} is needed for the white-wine fault")
    white = pd.read_csv(source, sep=";").sample(len(df), replace=True, random_state=5)
    white = white.reset_index(drop=True)
    if "wine_id" in df.columns:
        white["wine_id"] = df["wine_id"].to_numpy()
    return white[df.columns]


def restore_column(df: pd.DataFrame, **_) -> pd.DataFrame:
    return df


===== harness/faults/model.py =====
"""Model-layer faults (owner P4, roadmap 10.6 and 13.4).

Each inject function takes one replayed batch (wine_id, 11 features in
snake_case, quality), the index of the batch inside the fault's duration
(0-based) and the fault's ``target``; it returns the changed batch. They
change labels (what tasters say) or which rows reach a training window,
never the live inputs, so the Data Checker cannot see them.
"""

from __future__ import annotations

import numpy as np
import pandas as pd


def _mask(batch: pd.DataFrame, target: dict) -> pd.Series:
    mask = pd.Series(True, index=batch.index)
    if "alcohol_gt" in target:
        mask &= batch["alcohol"] > target["alcohol_gt"]
    if "alcohol_le" in target:
        mask &= batch["alcohol"] <= target["alcohol_le"]
    return mask


def shift_labels(batch: pd.DataFrame, step: int, target: dict) -> pd.DataFrame:
    """Concept drift: matching wines scored ``delta`` points differently.

    With ``ramp_batches`` the share of matching wines affected grows from
    1/ramp to 100% over that many batches (gradual drift).
    """
    out = batch.copy()
    mask = _mask(out, target)
    ramp = target.get("ramp_batches")
    if ramp:
        rng = np.random.default_rng(target.get("seed", 0) + step)
        mask &= rng.random(len(out)) < min(1.0, (step + 1) / ramp)
    out.loc[mask, "quality"] = out.loc[mask, "quality"] + target["delta"]
    return out


def drop_rows(batch: pd.DataFrame, step: int, target: dict) -> pd.DataFrame:
    """Group damage: matching wines never reach the batch (e.g. high-alcohol wines lost upstream)."""
    return batch[~_mask(batch, target)].copy()


def shuffle_labels(batch: pd.DataFrame, step: int, target: dict) -> pd.DataFrame:
    """Corrupted window: labels shuffled inside the batch (bad candidate if trained on it)."""
    out = batch.copy()
    rng = np.random.default_rng(target.get("seed", 0) + step)
    out["quality"] = rng.permutation(out["quality"].to_numpy())
    return out


def restore(batch: pd.DataFrame, step: int, target: dict) -> pd.DataFrame:
    """Revert: the replay writes clean batches again once the duration is over."""
    return batch


===== harness/replay.py =====
"""
Replay script (owner P3, roadmap 9.1 and 13.6).

Draws batches of 100 wines with replacement from the red-wine data, one per
simulated hour. For each batch it
  1. saves the file and adds a `pending` row to `batches`,
  2. sends the same wines (features only, with wine_id) to the API as live requests,
  3. writes their true quality to `labels`, available after labels.delay.
A data fault is applied to the feed, so it reaches the batch and the live requests.

Source: data/winequality-red.csv if present, else samples/feature_sample.parquet.
"""
from __future__ import annotations

from datetime import timedelta
from pathlib import Path
from typing import Callable, Optional

import numpy as np
import pandas as pd

from shared import clock

ROOT = Path(__file__).resolve().parents[1]
LABEL = "quality"


def load_source() -> pd.DataFrame:
    csv = ROOT / "data" / "winequality-red.csv"
    if csv.exists():
        return pd.read_csv(csv, sep=";")
    return pd.read_parquet(ROOT / "samples" / "feature_sample.parquet")


def http_sender(api_url: str) -> Callable[[dict], None]:
    """Sends one replayed wine to the baseline API (list of 11 numbers, as today)."""
    import httpx
    client = httpx.Client(base_url=api_url, timeout=5)

    def send(features: dict) -> None:
        values = [v for k, v in features.items() if k != "wine_id"]
        client.post("/predict", json={"data": values, "wine_id": features.get("wine_id")})
    return send


class Replay:
    def __init__(self, tables, batch_dir: Path, source: Optional[pd.DataFrame] = None,
                 batch_size: int = 100, seed: int = 0, label_delay: timedelta = timedelta(days=1),
                 batch_every: timedelta = timedelta(hours=1), fault_runner=None,
                 send_request: Optional[Callable[[dict], None]] = None):
        self.tables = tables
        self.batch_dir = Path(batch_dir)
        self.batch_dir.mkdir(parents=True, exist_ok=True)
        self.source = (source if source is not None else load_source()).reset_index(drop=True)
        self.batch_size = batch_size
        self.rng = np.random.default_rng(seed)
        self.label_delay = label_delay
        self.batch_every = batch_every
        self.fault_runner = fault_runner
        self.send_request = send_request
        self.index = 0
        self.next_wine_id = 1
        self.sent_requests = 0

    def next_batch(self) -> str:
        """Advances the clock one batch period and publishes the next batch."""
        if clock.mode() == "frozen":
            clock.advance(self.batch_every)
        self.index += 1
        batch_id = f"batch-{self.index:04d}"
        rows = self.source.iloc[self.rng.integers(0, len(self.source), self.batch_size)].copy()
        rows.insert(0, "wine_id", range(self.next_wine_id, self.next_wine_id + len(rows)))
        self.next_wine_id += len(rows)
        rows = rows.reset_index(drop=True)
        if self.fault_runner:      # model faults (concept drift): what tasters say changes, not the inputs
            rows = self.fault_runner.apply(self.index, rows, "labels", batch_id)

        batch = rows
        if self.fault_runner:
            batch = self.fault_runner.apply(self.index, rows, "batch", batch_id)
        path = self.batch_dir / f"{batch_id}.parquet"
        batch.to_parquet(path, index=False)
        self.tables.add_batch(batch_id, str(path), len(batch))

        requests = rows.drop(columns=[LABEL], errors="ignore")
        if self.fault_runner:
            requests = self.fault_runner.apply(self.index, requests, "requests", batch_id)
        if self.send_request:
            for record in requests.to_dict("records"):
                self.send_request(record)
        self.sent_requests += len(requests)

        available = (clock.now() + self.label_delay).isoformat().replace("+00:00", "Z")
        self.tables.add_labels([(w, q, available) for w, q in zip(rows["wine_id"], rows[LABEL])])
        return batch_id


===== harness/scenario_runner.py =====
"""
Scenario runner (owner P3, roadmap 13 and 14). Runs scenarios/<id>.yaml with the
real agents asked for and scripted fakes for the others, then checks the
expectations: messages that must appear (any order), alternative paths
("either"), messages that must not appear, human cards, end state.

    python -m harness.scenario_runner C                 # solo: real checker, fakes for the rest
    python -m harness.scenario_runner C --agents checker
"""
from __future__ import annotations

import argparse
import contextlib
import io
import logging
import sys
import tempfile
from dataclasses import dataclass, field
from datetime import datetime, timezone
from pathlib import Path

import yaml

from shared import clock
from shared.tables import Tables
from shared.testing import InMemoryBoard, SimpleIncidents, inprocess_dry_run
from harness.fakes import FakeAgent
from harness.fault_runner import FaultRunner, all_faults
from harness.replay import Replay

ROOT = Path(__file__).resolve().parents[1]
SCENARIO_DIR = ROOT / "scenarios"
ALL_AGENTS = ["setup", "reviewer", "checker", "trainer", "deployer", "monitor"]


def real_agent(name: str, board, incidents, tables, data_dir: Path):
    if name == "checker":
        from agents.checker.agent import CheckerAgent
        from agents.checker.settings import CheckerSettings
        return CheckerAgent(board=board, incidents=incidents, tables=tables,
                            settings=CheckerSettings(data_dir=data_dir / "checker"),
                            dry_run=inprocess_dry_run)
    raise NotImplementedError(f"agent {name!r} is not merged yet: run it as a fake")


def matches(msg, spec: dict) -> bool:
    if msg.kind != spec["kind"]:
        return False
    if "sender" in spec and msg.sender != spec["sender"]:
        return False
    if "type" in spec and msg.type.value != spec["type"]:
        return False
    if "to" in spec and not set(spec["to"]) <= set(msg.to):
        return False
    return all(msg.payload.get(k) == v for k, v in spec.get("payload", {}).items())


@dataclass
class ScenarioResult:
    scenario: str
    passed: bool
    problems: list[str] = field(default_factory=list)
    messages: list = field(default_factory=list)
    batches: dict = field(default_factory=dict)

    def summary(self) -> str:
        head = f"scenario {self.scenario}: {'PASS' if self.passed else 'FAIL'}"
        return "\n".join([head] + [f"  - {p}" for p in self.problems])


def evaluate(scenario: dict, board, tables) -> list[str]:
    expect, problems = scenario.get("expect", {}), []
    msgs = board.messages
    for spec in expect.get("messages", []):
        if not any(matches(m, spec) for m in msgs):
            problems.append(f"missing message {spec}")
    either = expect.get("either")
    if either and not any(all(any(matches(m, s) for m in msgs) for s in group) for group in either):
        problems.append(f"none of the alternative paths happened: {either}")
    for spec in expect.get("must_not", []):
        bad = [m for m in msgs if matches(m, spec)]
        if bad:
            problems.append(f"forbidden message {spec} seen {len(bad)}x")
    cards = {m.incident_id for m in msgs if m.kind == "request.approval"}
    if len(cards) > expect.get("max_human_cards", 99):
        problems.append(f"{len(cards)} human cards > {expect['max_human_cards']}")
    for batch_id, status in scenario.get("end_state", {}).get("batches", {}).items():
        row = tables.batch(batch_id)
        if not row or row["status"] != status:
            problems.append(f"batch {batch_id} is {row and row['status']}, expected {status}")
    return problems


def run_scenario(scenario_id: str, real: list[str] | None = None, verbose: bool = False) -> ScenarioResult:
    scenario = yaml.safe_load((SCENARIO_DIR / f"{scenario_id}.yaml").read_text(encoding="utf-8"))
    real = real or [scenario.get("expect", {}).get("owner", "checker")]
    fault = all_faults()[scenario["fault"]]
    n_batches = scenario.get("run", {}).get("batches") or (
        fault.after_batches + int(scenario.get("deadline", {}).get("simulated_hours", 6)))

    clock.configure(mode="frozen", start=datetime(2026, 3, 14, tzinfo=timezone.utc))
    work = Path(tempfile.mkdtemp(prefix=f"scenario_{scenario_id}_"))
    board, incidents, tables = InMemoryBoard(), SimpleIncidents(), Tables(":memory:")
    replay = Replay(tables, work / "batches", seed=scenario.get("seed", 0),
                    fault_runner=FaultRunner([fault], tables, run_id=f"solo-{scenario_id}"))
    agents = [real_agent(n, board, incidents, tables, work) for n in real]
    stubs = scenario.get("stubs", {})
    fakes = [FakeAgent(n, board, incidents, stubs.get(n, {}).get("script"),
                       stubs.get(n, {}).get("answers")) for n in ALL_AGENTS if n not in real]
    for f in fakes:
        board.subscribe(f.name, stubs.get(f.name, {}).get("subscriptions", []))

    out = io.StringIO()
    with contextlib.redirect_stdout(out if not verbose else sys.stdout):
        for i in range(1, n_batches + 1):
            replay.next_batch()
            for f in fakes:
                f.post_scripted(i)
            for _ in range(2):                    # second round handles the answers
                for a in agents:
                    a.step()
                for f in fakes:
                    f.respond()

    problems = evaluate(scenario, board, tables)
    return ScenarioResult(scenario_id, not problems, problems, list(board.messages),
                          {b["batch_id"]: b["status"] for b in tables.batches()})


def main() -> int:
    p = argparse.ArgumentParser()
    p.add_argument("scenario")
    p.add_argument("--agents", nargs="*", help="real agents (default: the scenario owner)")
    p.add_argument("-v", "--verbose", action="store_true")
    args = p.parse_args()
    logging.basicConfig(level=logging.INFO if args.verbose else logging.WARNING)
    result = run_scenario(args.scenario, args.agents, args.verbose)
    for m in result.messages:
        print(f"{m.time}  {m.sender:9s} -> {','.join(m.to) or '*':28s} {m.kind:24s} {m.summary}")
    print(result.summary())
    return 0 if result.passed else 1


if __name__ == "__main__":
    sys.exit(main())

===== shared/base_agent.py =====
"""
Base agent (owner P3): the 7-step loop every agent runs (roadmap 3.2 and 5.8).

    1 observe   -> 2 analyze -> 3 decide (LLM, rule fallback) -> 4 dry_run (level 2)
    -> 5 act (level 3 asks a human) -> 6 verify -> 7 remember

Each agent subclasses BaseAgent and writes six methods: observe, analyze,
context, rule_decision, act, handle. The loop, the action levels, the dry-run
gate, idempotent message handling, logging and memory are written once here,
so no agent can skip them.

The board (P6), LLM helper and dry-run runner (P4), experience log (P5) and
incidents are passed in. Until the real ones exist, the in-memory stand-ins
from shared/testing.py are used, so every agent can run alone.

Two message formats meet here: boards that store plain dicts (shared/standins.py
SqliteBoard, used between processes) and our in-memory board of Message objects.
step() hands our agents a Message, posts go out as dicts, and reply() answers a
message in the format its board gave it. Agents that keep their own loop (the
Trainer) can still subclass BaseAgent with dict menus and dict messages.
"""
from __future__ import annotations

import dataclasses
import inspect
import logging
import time
from dataclasses import dataclass, field
from typing import Any, Callable, Optional

from shared import clock
from shared.messages import Message, MessageType

log = logging.getLogger("agents")

LEVEL_INFO, LEVEL_SAFE, LEVEL_DRY_RUN, LEVEL_HUMAN = 0, 1, 2, 3

# shared/llm.py reports why it fell back; for us all of them are a fallback to the rules
FALLBACK_SOURCES = {"llm_invalid": "fallback", "llm_error": "fallback"}


def as_message(msg: Any) -> Message:
    """A board message, dict or Message, as a Message."""
    if isinstance(msg, Message):
        return msg
    return Message(**{k: v for k, v in msg.items() if k in Message.model_fields})


def message_id(msg: Any) -> str:
    return msg["id"] if isinstance(msg, dict) else msg.id


@dataclass(frozen=True)
class Action:
    """One entry of an agent's menu. The LLM may only pick actions from the menu."""
    name: str
    level: int                                   # 0 info, 1 safe, 2 dry run first, 3 human
    params_schema: dict = field(default_factory=dict)
    success_check: str = ""                      # written before acting, checked by verify()
    dry_run_kind: Optional[str] = None           # dry_run_jobs/<kind>.py, required for level 2
    safest: bool = False                         # the fallback when a dry run fails


@dataclass
class Analysis:
    """Output of step 2: plain numbers, a yes/no flag and the dependency-map nodes involved."""
    flag: bool
    nodes: list[str] = field(default_factory=list)
    numbers: dict = field(default_factory=dict)
    problem: str = ""                            # short text, used to find similar past cases


@dataclass
class Decision:
    action: str
    level: int
    params: dict = field(default_factory=dict)
    reason: str = ""
    confidence: Optional[float] = None
    success_check: str = ""
    memory_refs: list[str] = field(default_factory=list)
    source: str = "rules"                        # rules | llm | fallback


@dataclass
class Result:
    ok: bool
    outputs: dict = field(default_factory=dict)
    detail: str = ""


@dataclass
class StepReport:
    """What one round of the loop did. Returned by step(); useful in tests and logs."""
    handled: list[str] = field(default_factory=list)
    flagged: bool = False
    incident_id: Optional[str] = None
    decision: Optional[Decision] = None
    dry_run_ok: Optional[bool] = None
    waiting_for_human: bool = False
    result: Optional[Result] = None
    verified: Optional[bool] = None
    memory_id: Optional[str] = None


def to_decision(answer: Any) -> Optional[Decision]:
    """What an LLM helper returns (our Decision, a dict, or shared/llm.py's pydantic Decision)
    as our Decision; None stays None so valid() rejects it."""
    if answer is None or isinstance(answer, Decision):
        return answer
    data = answer.model_dump() if hasattr(answer, "model_dump") else dict(answer)
    known = {f.name for f in dataclasses.fields(Decision)}
    decision = Decision(**{k: v for k, v in data.items() if k in known})
    decision.source = FALLBACK_SOURCES.get(decision.source, decision.source)
    return decision


class BaseAgent:
    name: str = "base"
    menu: list[Action] = []
    subscriptions: list[str] = []                # topic globs, e.g. "dataset:*"

    def __init__(self, board=None, llm=None, dry_run: Optional[Callable] = None,
                 memory=None, incidents=None, action_log=None,
                 reason: str = "llm", window_s: float = 3600.0):
        from shared import testing                # stand-ins until the real pieces exist
        self.board = board or testing.InMemoryBoard()
        self.llm = llm or testing.FakeLLM()
        self.dry_run = dry_run or testing.fake_dry_run
        self.memory = memory or testing.InMemoryMemory()
        self.incidents = incidents or testing.SimpleIncidents()
        self.action_log = action_log or testing.ActionLog()
        self.reason = reason                     # "llm", or "rules" for the traditional system
        self.window_s = window_s
        self.inbox: list[Message] = []
        self.incident_id: Optional[str] = None   # incident of the current round, for act()
        self._handled_ids: set[str] = set()      # ids of the messages already handled
        self._raw: dict[str, Any] = {}           # message id -> the message as its board gave it
        for a in self.menu:
            if isinstance(a, Action) and a.level == LEVEL_DRY_RUN and not a.dry_run_kind:
                raise ValueError(f"{self.name}: level-2 action {a.name!r} needs a dry_run_kind")
        if hasattr(self.board, "subscribe"):
            self.board.subscribe(self.name, self.subscriptions)

    # ---- each agent writes these --------------------------------------------
    def observe(self) -> dict:
        """1. Look: read inputs (new batch, last hour of predictions, latest commit)."""
        raise NotImplementedError

    def analyze(self, obs: dict) -> Analysis:
        """2. Understand: statistics and checks, no LLM. No flag means nothing to do."""
        raise NotImplementedError

    def context(self, analysis: Analysis, inbox: list[Message]) -> dict:
        """What the LLM sees: numbers and memory matches, never raw rows."""
        raise NotImplementedError

    def rule_decision(self, analysis: Analysis) -> Decision:
        """Fallback when the LLM fails AND the decision of the 'rules only' system."""
        raise NotImplementedError

    def act(self, decision: Decision) -> Result:
        """5. Do: call the function of actions.py for this decision."""
        raise NotImplementedError

    def handle(self, msg: Message) -> None:
        """Questions and requests from other agents."""
        raise NotImplementedError

    def verify(self, decision: Decision, result: Result) -> bool:
        """6. Check. Override to test the action's success_check for real."""
        return result.ok

    # ---- given by the base class ----------------------------------------------
    def step(self) -> StepReport:
        report = StepReport()
        self.inbox = []
        self.incident_id = None
        for raw in self.board.read(self.name):
            try:
                msg = as_message(raw)
            except Exception:                    # a malformed message must not stop the loop
                log.exception("%s: unreadable message %r", self.name, raw)
                continue
            self._raw[msg.id] = raw
            if self.safe_handle(msg):
                report.handled.append(msg.id)

        analysis = self.analyze(self.observe())                          # 1 + 2
        if not analysis.flag:
            self.board.heartbeat(self.name)
            return report
        report.flagged = True
        incident = self.incidents.open_or_join(analysis.nodes, self.window_s)
        report.incident_id = self.incident_id = incident

        decision = self.decide(analysis)                                 # 3
        if decision.level == LEVEL_DRY_RUN:                              # 4
            action = self.action(decision.action)
            outcome = self.dry_run(action.dry_run_kind, {"action": decision.action,
                                                         "params": decision.params})
            report.dry_run_ok = bool(outcome.ok)
            if not outcome.ok:
                log.info("%s: dry run of %s failed, falling back to the safest action",
                         self.name, decision.action)
                decision = self.safest(analysis, reason=f"dry run of {decision.action} failed")
        report.decision = decision

        if decision.level == LEVEL_HUMAN:                                # 5 (level 3)
            self.ask_human(incident, decision)
            report.waiting_for_human = True
            return report

        started = time.monotonic()
        result = self.act(decision)                                      # 5
        ok = self.verify(decision, result)                               # 6
        report.result, report.verified = result, ok
        report.memory_id = self.memory.remember(                         # 7
            agent=self.name, incident_id=incident, problem=analysis.problem,
            evidence=analysis.numbers, action=decision.action, params=decision.params,
            level=decision.level, outcome="worked" if ok else "failed",
            time_to_recover_s=time.monotonic() - started)
        self.action_log.action(self.name, incident, decision, result, ok)
        return report

    def decide(self, analysis: Analysis) -> Decision:
        """3. The LLM picks one menu action; any problem -> the rule decision."""
        fallback = self.rule_decision(analysis)
        fallback.source = "rules"
        if self.reason == "rules":
            return fallback
        kwargs = {}
        if "incident_id" in inspect.signature(self.llm.decide).parameters:
            kwargs["incident_id"] = self.incident_id       # shared/llm.py logs it in llm_calls
        try:
            answer = self.llm.decide(self, self.context(analysis, self.inbox), self.menu,
                                     fallback=dataclasses.asdict(fallback), **kwargs)
            decision = to_decision(answer)
        except Exception:                        # the loop must never stop on the LLM
            log.exception("%s: llm.decide failed, using the rule decision", self.name)
            return fallback
        if not self.valid(decision):
            log.warning("%s: invalid LLM decision %r, using the rule decision", self.name, decision)
            fallback.source = "fallback"
            return fallback
        return decision

    def valid(self, decision: Optional[Decision]) -> bool:
        """The action must exist in the menu, with the menu's level."""
        if decision is None:
            return False
        try:
            return self.action(decision.action).level == decision.level
        except KeyError:
            return False

    def action(self, name: str) -> Action:
        for a in self.menu:
            if a.name == name:
                return a
        raise KeyError(f"{self.name} has no action {name!r} in its menu")

    def safest(self, analysis: Analysis, reason: str = "") -> Decision:
        """The menu action marked safest, else the lowest-level one."""
        candidates = [a for a in self.menu if a.safest] or sorted(self.menu, key=lambda a: a.level)
        a = candidates[0]
        return Decision(a.name, a.level, {}, reason=reason, success_check=a.success_check,
                        source="fallback")

    def safe_handle(self, msg: Any) -> bool:
        """Handles each message once, and never lets a bad message crash the loop.
        Takes a Message, or a dict for agents that keep their own loop (the Trainer)."""
        mid = message_id(msg)
        if mid in self._handled_ids:
            return False
        self._handled_ids.add(mid)
        if isinstance(msg, Message):
            self.inbox.append(msg)
        try:
            self.handle(msg)
        except Exception:
            kind = msg.get("kind") if isinstance(msg, dict) else msg.kind
            log.exception("%s: error while handling %s (%s)", self.name, mid, kind)
        return True

    def reply(self, msg: Message, **fields: Any) -> str:
        """Answers a message through the board, in the format the board gave it."""
        original = self._raw.get(msg.id, msg)
        fields.setdefault("sender", self.name)
        fields.setdefault("time", clock.iso())
        return self.board.answer(original, **fields)

    def ask_human(self, incident_id: Optional[str], decision: Decision) -> str:
        """Level 3: one request.approval card; the human answers through the CLI."""
        return self.post(
            type=MessageType.REQUEST, kind="request.approval", to=["human"],
            topic=f"incident:{incident_id}", incident_id=incident_id, level=LEVEL_HUMAN,
            summary=f"{self.name} asks approval for {decision.action}"[:160],
            evidence={"reason": decision.reason},
            payload={"question": f"Approve {decision.action}?", "options": ["approve", "reject"],
                     "default_if_no_answer": "reject", "card_ref": incident_id,
                     "action": decision.action, "params": decision.params})

    def post(self, *, type: MessageType, kind: str, topic: str, summary: str,
             to: Optional[list[str]] = None, level: int = LEVEL_INFO, payload: Optional[dict] = None,
             evidence: Optional[dict] = None, **fields: Any) -> str:
        """Builds a message from this agent with the shared clock's time, and posts it."""
        msg = Message(type=type, kind=kind, sender=self.name, to=to or [], topic=topic,
                      summary=summary, level=level, payload=payload or {},
                      evidence=evidence or {}, time=clock.iso(), **fields)
        return self.board.post(msg.model_dump(mode="json"))     # boards store plain dicts

    def run(self, poll_s: float = 2.0, max_steps: Optional[int] = None) -> None:
        """Main loop: one step every poll_s seconds (2 s in integration mode)."""
        n = 0
        while max_steps is None or n < max_steps:
            try:
                self.step()
            except Exception:                    # one bad round must not stop the agent
                log.exception("%s: step failed", self.name)
            n += 1
            time.sleep(poll_s)


===== shared/clock.py =====
"""
Shared clock (owner P3). Every agent takes the time from clock.now(), never
from datetime.now(), so a replayed day behaves the same for everyone.

Three modes:
- real:   real UTC time.
- replay: simulated time = start + (wall time - epoch) * speed. It only depends
          on the wall clock and on CLOCK_START / CLOCK_EPOCH / CLOCK_SPEED, so
          every agent process computes the same simulated time.
- frozen: time moves only through advance() / set_time(). For unit tests.

Configuration: configure(...) in code, or the environment variables
CLOCK_MODE, CLOCK_SPEED (default 720: 1 real second = 12 simulated minutes),
CLOCK_START (ISO-8601, simulated start) and CLOCK_EPOCH (unix seconds, real
moment the replay started). The harness sets them before starting the agents.
"""
from __future__ import annotations

import os
import time
from datetime import datetime, timedelta, timezone

DEFAULT_SPEED = 720.0
MODES = ("real", "replay", "frozen")

_state: dict = {}


def _parse(value: str) -> datetime:
    dt = datetime.fromisoformat(value.replace("Z", "+00:00"))
    return dt if dt.tzinfo else dt.replace(tzinfo=timezone.utc)


def configure(mode: str | None = None, speed: float | None = None,
              start: datetime | str | None = None, epoch: float | None = None) -> None:
    """Sets the clock. Arguments left out are read from the environment."""
    mode = mode or os.environ.get("CLOCK_MODE", "real")
    if mode not in MODES:
        raise ValueError(f"clock mode must be one of {MODES}, got {mode!r}")
    start = start or os.environ.get("CLOCK_START")
    if isinstance(start, str):
        start = _parse(start)
    if start is None:
        start = datetime.now(timezone.utc)
    if epoch is None:
        epoch = float(os.environ.get("CLOCK_EPOCH", time.time()))
    speed = float(speed if speed is not None else os.environ.get("CLOCK_SPEED", DEFAULT_SPEED))
    _state.clear()
    _state.update(mode=mode, speed=speed, start=start, epoch=epoch, offset=timedelta(0))


def now() -> datetime:
    """Current time, always timezone-aware UTC."""
    if not _state:
        configure()
    mode = _state["mode"]
    if mode == "real":
        return datetime.now(timezone.utc) + _state["offset"]
    if mode == "frozen":
        return _state["start"] + _state["offset"]
    elapsed = (time.time() - _state["epoch"]) * _state["speed"]
    return _state["start"] + timedelta(seconds=elapsed) + _state["offset"]


def iso() -> str:
    """now() as an ISO-8601 string ending in Z, the format used in messages."""
    return now().isoformat().replace("+00:00", "Z")


def advance(delta: timedelta | float) -> datetime:
    """Moves the clock forward (seconds or timedelta). Used by the harness and tests."""
    if not _state:
        configure()
    if not isinstance(delta, timedelta):
        delta = timedelta(seconds=delta)
    _state["offset"] += delta
    return now()


def set_time(value: datetime | str) -> datetime:
    """Jumps to an exact time, keeping the current mode."""
    target = _parse(value) if isinstance(value, str) else value
    _state["offset"] += target - now()
    return now()


def mode() -> str:
    if not _state:
        configure()
    return _state["mode"]


def env_for_agents() -> dict:
    """Environment variables that make another process share this replay clock."""
    if not _state:
        configure()
    start = _state["start"] + _state["offset"]
    return {"CLOCK_MODE": _state["mode"], "CLOCK_SPEED": str(_state["speed"]),
            "CLOCK_START": start.isoformat(), "CLOCK_EPOCH": str(_state["epoch"])}


===== shared/depmap.py =====
from __future__ import annotations
from dataclasses import dataclass, field
from pathlib import Path
from typing import Dict, List, Optional, Set
import yaml


@dataclass
class DependencyLink:
    src: str
    dst: str
    relation: str


class DependencyMap:
    """
    Dependency map tracking upstream/downstream relationships between:
    - code:<path>
    - feature:<name>
    - dataset:<batch_id>
    - model:<name>@<version>
    - deployment:<target>
    """

    def __init__(self, data: Optional[dict] = None):
        self._raw_data = data or {}
        self._nodes: Set[str] = set()
        self._owners: Dict[str, str] = {}
        self._links: List[DependencyLink] = []
        if data:
            self._parse(data)

    def _parse(self, data: dict):
        # Parse owners
        for node, owner in data.get("owners", {}).items():
            self._nodes.add(node)
            self._owners[node] = owner

        # Parse links
        for link in data.get("links", []):
            src = link["src"]
            dst = link["dst"]
            relation = link.get("relation", "depends_on")
            self._nodes.add(src)
            self._nodes.add(dst)
            self._links.append(DependencyLink(src=src, dst=dst, relation=relation))

    @classmethod
    def from_yaml(cls, path: str | Path) -> DependencyMap:
        with open(path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f) or {}
        return cls(data)

    def to_dict(self) -> dict:
        return {
            "owners": dict(self._owners),
            "links": [
                {"src": l.src, "dst": l.dst, "relation": l.relation}
                for l in self._links
            ],
        }

    def save_yaml(self, path: str | Path) -> None:
        with open(path, "w", encoding="utf-8") as f:
            yaml.safe_dump(self.to_dict(), f, sort_keys=False)

    def upstream(self, node: str) -> List[str]:
        """
        Returns all nodes that directly feed into or are upstream of this node.
        If src -> dst, then src is upstream of dst.
        """
        return [l.src for l in self._links if l.dst == node]

    def downstream(self, node: str) -> List[str]:
        """
        Returns all nodes that depend directly on this node.
        If src -> dst, then dst is downstream of src.
        """
        return [l.dst for l in self._links if l.src == node]

    def owner(self, node: str) -> Optional[str]:
        """Returns the assigned owner agent for a given node."""
        if node in self._owners:
            return self._owners[node]
        # Infer owner by node prefix if not explicitly set
        prefix = node.split(":")[0] if ":" in node else node
        prefix_to_owner = {
            "code": "reviewer",
            "feature": "reviewer",
            "dataset": "checker",
            "model": "trainer",
            "deployment": "deployer",
            "pipeline": "setup",
        }
        return prefix_to_owner.get(prefix)

    def add_link(self, src: str, dst: str, relation: str = "depends_on") -> None:
        """Adds a directional link src -> dst."""
        self._nodes.add(src)
        self._nodes.add(dst)
        for link in self._links:
            if link.src == src and link.dst == dst:
                link.relation = relation
                return
        self._links.append(DependencyLink(src=src, dst=dst, relation=relation))

    def set_owner(self, node: str, owner: str) -> None:
        """Sets the owner for a node."""
        self._nodes.add(node)
        self._owners[node] = owner

    def walk_upstream(self, node: str) -> Set[str]:
        """Recursively traverses all upstream ancestor nodes."""
        visited: Set[str] = set()
        queue = [node]
        while queue:
            curr = queue.pop(0)
            for up in self.upstream(curr):
                if up not in visited:
                    visited.add(up)
                    queue.append(up)
        return visited


===== shared/dry_run.py =====
"""Dry-run runner (roadmap 5.10, owner P4).

    dry_run(kind, inputs, timeout_s=120) -> DryRunResult(ok, outputs, metrics, log_tail, seconds)

Runs ``dry_run_jobs/<kind>.py`` in a subprocess, inside a fresh temp folder,
with ``DRY_RUN=1`` and ``MLFLOW_EXPERIMENT_NAME=dry_run`` set. It gives
isolation, not logic: each agent writes its own job.

A job module defines one function:

    def run(inputs: dict, workdir: pathlib.Path) -> dict:
        return {"ok": True, "outputs": {...}, "metrics": {...}}

Anything the job prints ends up in ``log_tail``. An exception, a timeout or
a missing ``ok`` key all give ``ok=False``.

Mode: ``subprocess`` (default, used in solo/integration/experiment runs) or
``inprocess`` (unit tests: same contract, no subprocess). Pick it with the
``mode`` argument or the ``DRY_RUN_MODE`` environment variable.
"""

from __future__ import annotations

import contextlib
import importlib.util
import io
import json
import os
import subprocess
import sys
import tempfile
import time
import traceback
from dataclasses import dataclass, field
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
JOBS_DIR = ROOT / "dry_run_jobs"
LOG_TAIL_LINES = 50
DRY_RUN_EXPERIMENT = "dry_run"


@dataclass
class DryRunResult:
    ok: bool
    outputs: dict = field(default_factory=dict)
    metrics: dict = field(default_factory=dict)
    log_tail: str = ""
    seconds: float = 0.0


class DryRunViolation(RuntimeError):
    """Raised when code tries to write real state during a dry run."""


def is_dry_run() -> bool:
    return os.environ.get("DRY_RUN") == "1"


def guard_real_write(what: str) -> None:
    """Call before writing a real table, alias or file. shared/tables.py uses this."""
    if is_dry_run():
        raise DryRunViolation(f"refused to write {what!r} while DRY_RUN=1")


def _job_path(kind: str, jobs_dir: Path) -> Path:
    if not kind.replace("_", "").isalnum():
        raise ValueError(f"bad dry-run kind {kind!r}")
    return jobs_dir / f"{kind}.py"


def _tail(text: str, n: int = LOG_TAIL_LINES) -> str:
    return "\n".join(text.splitlines()[-n:])


def _as_result(raw: object, log: str, seconds: float) -> DryRunResult:
    if not isinstance(raw, dict) or "ok" not in raw:
        return DryRunResult(False, log_tail=_tail(log + f"\njob returned {raw!r}, expected a dict with 'ok'"), seconds=seconds)
    return DryRunResult(
        ok=bool(raw["ok"]),
        outputs=dict(raw.get("outputs") or {}),
        metrics=dict(raw.get("metrics") or {}),
        log_tail=_tail(log),
        seconds=seconds,
    )


def dry_run(
    kind: str,
    inputs: dict,
    timeout_s: float = 120,
    *,
    mode: str | None = None,
    jobs_dir: str | Path | None = None,
) -> DryRunResult:
    jobs_dir = Path(jobs_dir) if jobs_dir else JOBS_DIR
    job = _job_path(kind, jobs_dir)
    if not job.exists():
        return DryRunResult(False, log_tail=f"no dry-run job for kind {kind!r} at {job}")
    mode = mode or os.environ.get("DRY_RUN_MODE", "subprocess")
    if mode == "inprocess":
        return _run_inprocess(job, inputs)
    if mode != "subprocess":
        raise ValueError(f"unknown dry-run mode {mode!r}")
    return _run_subprocess(job, inputs, timeout_s)


def _run_subprocess(job: Path, inputs: dict, timeout_s: float) -> DryRunResult:
    start = time.monotonic()
    with tempfile.TemporaryDirectory(prefix=f"dry_run_{job.stem}_") as tmp:
        workdir = Path(tmp)
        (workdir / "inputs.json").write_text(json.dumps(inputs, default=str))
        env = {
            **os.environ,
            "DRY_RUN": "1",
            "MLFLOW_EXPERIMENT_NAME": DRY_RUN_EXPERIMENT,
            "PYTHONPATH": os.pathsep.join(filter(None, [str(ROOT), os.environ.get("PYTHONPATH")])),
        }
        cmd = [sys.executable, "-m", "shared._dry_run_entry", str(job), str(workdir)]
        try:
            proc = subprocess.run(cmd, cwd=workdir, env=env, capture_output=True, text=True, timeout=timeout_s)
        except subprocess.TimeoutExpired as exc:
            parts = (exc.stdout, exc.stderr)
            log = "".join(p.decode(errors="replace") if isinstance(p, bytes) else p for p in parts if p)
            return DryRunResult(False, log_tail=_tail(log + f"\ntimeout after {timeout_s}s"), seconds=time.monotonic() - start)
        log = proc.stdout + proc.stderr
        seconds = time.monotonic() - start
        result_file = workdir / "result.json"
        if proc.returncode != 0 or not result_file.exists():
            return DryRunResult(False, log_tail=_tail(log + f"\nexit code {proc.returncode}"), seconds=seconds)
        return _as_result(json.loads(result_file.read_text()), log, seconds)


@contextlib.contextmanager
def _dry_env():
    saved = {k: os.environ.get(k) for k in ("DRY_RUN", "MLFLOW_EXPERIMENT_NAME")}
    os.environ["DRY_RUN"] = "1"
    os.environ["MLFLOW_EXPERIMENT_NAME"] = DRY_RUN_EXPERIMENT
    try:
        yield
    finally:
        for k, v in saved.items():
            if v is None:
                os.environ.pop(k, None)
            else:
                os.environ[k] = v


def _run_inprocess(job: Path, inputs: dict) -> DryRunResult:
    start = time.monotonic()
    buf = io.StringIO()
    with tempfile.TemporaryDirectory(prefix=f"dry_run_{job.stem}_") as tmp, _dry_env():
        try:
            with contextlib.redirect_stdout(buf), contextlib.redirect_stderr(buf):
                raw = load_job(job).run(inputs, Path(tmp))
        except Exception:
            return DryRunResult(False, log_tail=_tail(buf.getvalue() + traceback.format_exc()), seconds=time.monotonic() - start)
    return _as_result(raw, buf.getvalue(), time.monotonic() - start)


def load_job(path: Path):
    spec = importlib.util.spec_from_file_location(f"dry_run_jobs.{path.stem}", path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    if not hasattr(module, "run"):
        raise AttributeError(f"{path} has no run(inputs, workdir) function")
    return module


===== shared/llm.py =====
"""LLM helper (roadmap 5.9, owner P4).

    llm.decide(agent, context, menu, fallback) -> Decision

The only way an agent calls an LLM. The model must answer one action from
the menu, in JSON:

    {"action": "quarantine", "level": 1, "params": {"batch_id": "batch-0042"},
     "reason": "...", "confidence": 0.86, "success_check": "...", "memory_refs": ["mem-0012"]}

Rules:
- temperature 0, JSON output enforced (Ollama ``format: json``);
- the answer is checked with pydantic and against the menu: unknown action,
  wrong level or missing params -> one retry with the error added; a second
  failure -> the rule decision (``fallback``), logged as ``llm_invalid``;
- a provider error (timeout, connection refused) -> the rule decision, logged
  as ``llm_error``;
- the context holds numbers and memory matches only, capped near 2,000 tokens;
- every call is logged to the ``llm_calls`` table.

Provider comes from ``llm.provider`` in agents.yaml (or ``LLM_PROVIDER``):
``ollama`` (free, local, the default) | ``nvidia_nim`` / ``groq`` / ``openai`` (any
OpenAI-style endpoint, key read from ``llm.api_key_env``) | ``fake`` | ``off``. ``fake`` answers from a script
(unit tests); ``off`` skips the LLM and returns the rule decision, which is
the "traditional pipeline" setting.
"""

from __future__ import annotations

import json
import re
import os
from contextlib import closing
import sqlite3
import time
import uuid
from collections.abc import Callable, Iterable
from dataclasses import dataclass, field

    "model": "qwen2.5:3b",

}  # timeout_s: 60 for ollama, 300 for NVIDIA NIM (its free tier can queue); LLM_TIMEOUT_S overrides


# --------------------------------------------------------------------------
# Data shapes
# --------------------------------------------------------------------------
class Decision(BaseModel):
    action: str
    level: int = Field(ge=0, le=3)
    params: dict[str, Any] = Field(default_factory=dict)
    reason: str = ""
    confidence: float | None = Field(default=None, ge=0, le=1)
    success_check: str = ""
    memory_refs: list[str] = Field(default_factory=list)
    # set by decide(), not by the model: llm | rules | llm_invalid | llm_error
    source: str = "llm"


@dataclass
class MenuItem:
    """What decide() needs to know about one allowed action.

    Built from shared.base_agent.Action (name, level, params_schema,
    success_check) or from a plain dict with the same keys.
    """

    name: str
    level: int
    required_params: list[str] = field(default_factory=list)
    success_check: str = ""

    @classmethod
    def coerce(cls, item: Any) -> MenuItem:
        if isinstance(item, MenuItem):
            return item
        get = item.get if isinstance(item, dict) else (lambda k, d=None: getattr(item, k, d))
        schema = get("params_schema") or get("params") or {}
        if isinstance(schema, dict):
            required = schema.get("required", list(schema.get("properties", schema).keys()))
        else:
            required = list(schema)
        return cls(name=get("name"), level=int(get("level")), required_params=list(required), success_check=get("success_check") or "")

    error: str | None

class FakeLLM:
    """Scripted answers for unit tests and stubs.

    Each script entry is a dict (sent as JSON), a raw string, an Exception
    (raised), or a callable(messages) returning one of those. When the
    script runs out, it echoes the fallback decision.
    """

    def __init__(self, script: Iterable[Any] = ()):
        self.script = list(script)
        self.calls: list[list[dict]] = []
        self.fallback: dict | None = None

    def chat(self, messages: list[dict], cfg: dict) -> tuple[str, int | None, int | None]:
        self.calls.append(messages)
        answer: Any = self.script.pop(0) if self.script else self.fallback or {}
        if callable(answer):
            answer = answer(messages)
        if isinstance(answer, BaseException):
            raise answer
        text = answer if isinstance(answer, str) else json.dumps(answer)
        return text, None, None


class OllamaLLM:
    def chat(self, messages: list[dict], cfg: dict) -> tuple[str, int | None, int | None]:
        import httpx

        if not cfg.get("model"):
            raise LLMError("llm.model is not set in agents.yaml")
        resp = httpx.post(
            f"{cfg['base_url'].rstrip('/')}/api/chat",
            json={
                "model": cfg["model"],
                "messages": messages,
                "format": "json",
                "stream": False,
                "options": {"temperature": cfg.get("temperature", 0)},
            },
            timeout=float(cfg.get("timeout_s") or 60),
        )
        resp.raise_for_status()
        body = resp.json()
        return body["message"]["content"], body.get("prompt_eval_count"), body.get("eval_count")


class OpenAICompatLLM:
    """Any OpenAI-style /chat/completions endpoint, e.g. NVIDIA NIM (the Reviewer's choice).

    agents.yaml: llm.provider nvidia_nim, llm.base_url, llm.model and
    llm.api_key_env (the environment variable holding the key, NVIDIA_API_KEY by default).
    """

    def chat(self, messages: list[dict], cfg: dict) -> tuple[str, int | None, int | None]:
        import httpx

        key = os.environ.get(cfg.get("api_key_env") or "NVIDIA_API_KEY")
        if not key:
            raise LLMError(f"no API key in ${cfg.get('api_key_env') or 'NVIDIA_API_KEY'}")
        if not cfg.get("model"):
            raise LLMError("llm.model is not set in agents.yaml")
        base = cfg["base_url"].rstrip("/").removesuffix("/chat/completions")  # some .env files give the full URL
        url = f"{base}/chat/completions"
        body = {
            "model": cfg["model"],
            "messages": messages,
            "temperature": cfg.get("temperature", 0),
            "max_tokens": int(cfg.get("max_tokens") or 4096),  # room for a reasoning model's thinking
            "response_format": {"type": "json_object"},
        }
        headers = {"Authorization": f"Bearer {key}", "Accept": "application/json"}
        timeout = float(cfg.get("timeout_s") or 300)
        resp = httpx.post(url, headers=headers, json=body, timeout=timeout)
        if resp.status_code in (400, 422):  # some models refuse response_format; the prompt already asks for JSON
            body.pop("response_format")
            resp = httpx.post(url, headers=headers, json=body, timeout=timeout)
        if resp.status_code >= 400:
            raise LLMError(f"HTTP {resp.status_code} from {url}: {resp.text[:300]}")
        body = resp.json()
        usage = body.get("usage") or {}
        content = body["choices"][0]["message"].get("content") or ""
        return content, usage.get("prompt_tokens"), usage.get("completion_tokens")

def load_dotenv(path: str | Path | None = None) -> None:
    """Put KEY=VALUE lines of the repo's .env into os.environ (variables already set win). Keys stay out of git."""
    path = Path(path or ROOT / ".env")
    if not path.exists():
        return
    for line in path.read_text(encoding="utf-8-sig").splitlines():
        line = line.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        key, value = line.removeprefix("export ").split("=", 1)
        os.environ.setdefault(key.strip(), value.strip().strip('"').strip("'"))


def load_config(path: str | Path | None = None) -> dict:
    """llm section of agents.yaml + db_url + env overrides (LLM_PROVIDER, LLM_MODEL, LLM_BASE_URL, LLM_TIMEOUT_S)."""
    load_dotenv()
    cfg = dict(DEFAULT_CONFIG)
    db_url = os.environ.get("DB_URL")
    path = Path(path or os.environ.get("AGENTS_YAML", ROOT / "agents.yaml"))
    if path.exists():
        import yaml

        data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
        cfg.update(data.get("llm") or {})
        db_url = db_url or data.get("db_url")
    for env, key in (("LLM_PROVIDER", "provider"), ("LLM_MODEL", "model"), ("LLM_BASE_URL", "base_url"),
                     ("LLM_TIMEOUT_S", "timeout_s")):
        if os.environ.get(env):
            cfg[key] = os.environ[env]
    if cfg.get("provider") == "ollama" and os.environ.get("OLLAMA_BASE_URL"):  # e.g. host.docker.internal in Docker
        cfg["base_url"] = os.environ["OLLAMA_BASE_URL"]
    cfg["db_url"] = db_url
    return cfg


def config_for(provider: str) -> dict:
    """load_config(), switched to ``provider``; model and URL come from PROVIDER_DEFAULTS when agents.yaml names another one."""
    cfg = load_config()
    if provider in ("off", "fake") or provider == cfg.get("provider"):
        return {**cfg, "provider": provider}
    defaults = dict(PROVIDER_DEFAULTS.get(provider, {}))
    if provider == "groq":  # free key from console.groq.com/keys
        defaults["model"] = os.environ.get("GROQ_MODEL") or defaults["model"]
    if provider == "nvidia_nim":  # the Reviewer's .env names these
        defaults["model"] = os.environ.get("NVIDIA_MODEL") or defaults["model"]
        defaults["base_url"] = os.environ.get("NVIDIA_BASE_URL") or defaults["base_url"]
    if provider == "ollama":
        defaults["base_url"] = os.environ.get("OLLAMA_BASE_URL") or defaults.get("base_url")
    return {**cfg, **defaults, "provider": provider}


CALL_LOG: list[CallRecord] = []  # in-memory copy, handy in tests


def _sqlite_path(db_url: str | None) -> str | None:
    if db_url and db_url.startswith("sqlite:///"):
        return db_url[len("sqlite:///"):]
    return None


def _log_call(rec: CallRecord, db_url: str | None) -> None:
    CALL_LOG.append(rec)
    path = _sqlite_path(db_url)
    if not path:
        return
    with closing(sqlite3.connect(path, timeout=10)) as con, con:  # close too, so Windows can delete the file
        con.execute("PRAGMA journal_mode=WAL")
        con.execute(
            """CREATE TABLE IF NOT EXISTS llm_calls (
                id TEXT PRIMARY KEY, time TEXT, run_id TEXT, agent TEXT, incident_id TEXT,
                attempt INTEGER, prompt TEXT, raw_answer TEXT, parsed TEXT, outcome TEXT,
                error TEXT, tokens_in INTEGER, tokens_out INTEGER, latency_ms INTEGER,
                provider TEXT, model TEXT)"""
        )
        con.execute(
            "INSERT INTO llm_calls VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",
            (
                rec.id, _now(), os.environ.get("RUN_ID", "dev"), rec.agent, rec.incident_id,
                rec.attempt, rec.prompt, rec.raw_answer, json.dumps(rec.parsed) if rec.parsed else None,
                rec.outcome, rec.error, rec.tokens_in, rec.tokens_out, rec.latency_ms, rec.provider, rec.model,
            ),
        )


def _now() -> str:
    try:
        from shared import clock  # owner P3; replay time during experiments

    except Exception:

        return datetime.now(timezone.utc).isoformat()


# --------------------------------------------------------------------------
# Prompt building
# --------------------------------------------------------------------------
def cap_context(context: dict, max_tokens: int = MAX_CONTEXT_TOKENS) -> dict:
    """Shrink the context until it fits: drop items from the longest list first."""
    ctx = json.loads(json.dumps(context, default=str))
    limit = max_tokens * CHARS_PER_TOKEN

    def lists(obj, depth=0):
        if isinstance(obj, dict) and depth < 3:
            for v in obj.values():
                if isinstance(v, list) and v:
                    yield v
                yield from lists(v, depth + 1)

    while len(json.dumps(ctx)) > limit:
        candidates = sorted(lists(ctx), key=lambda lst: len(json.dumps(lst)), reverse=True)
        if not candidates:
            break
        candidates[0].pop()
        ctx["_truncated"] = True
    if len(json.dumps(ctx)) > limit:
        ctx = {"_truncated": True, "summary": json.dumps(ctx)[: limit - 100]}
    return ctx


def _agent_prompt(agent: Any) -> str:
    text = getattr(agent, "prompt", None)
    if text:
        return text
    name = getattr(agent, "name", None) or str(agent)
    path = ROOT / "agents" / name / "prompt.md"
    return path.read_text() if path.exists() else f"You are the {name} agent of a wine-quality MLOps pipeline."


def build_messages(agent: Any, context: dict, menu: list[MenuItem]) -> list[dict]:
    menu_text = "\n".join(
        f"- {m.name} (level {m.level}; params: {', '.join(m.required_params) or 'none'})"
        + (f"; success check: {m.success_check}" if m.success_check else "")
        for m in menu
    )
    system = (
        _agent_prompt(agent).strip()
        + "\n\nChoose exactly ONE action from this menu:\n"
        + menu_text
        + "\n\nAnswer with a single JSON object and nothing else, with keys: "
        '"action" (a menu name), "level" (the menu level of that action), "params" (object), '
        '"reason" (one sentence quoting the numbers you used), "confidence" (0.0-1.0), '
        '"success_check" (how to tell it worked), "memory_refs" (list of memory ids you relied on).'
    )
    return [
        {"role": "system", "content": system},
        {"role": "user", "content": json.dumps(cap_context(context), indent=1)},
    ]


def _json_part(raw: str | None) -> str:
    """The JSON object in an answer: drops a <think>...</think> block (Qwen3 and other reasoning models) and code fences."""
    text = re.sub(r"<think>.*?</think>", "", raw or "", flags=re.S).strip()
    start, end = text.find("{"), text.rfind("}")
    return text[start:end + 1] if start != -1 and end > start else text


def validate(raw: str, menu: list[MenuItem]) -> Decision:
    """Parse and check one answer. Raises ValueError with a message the model can act on."""
    try:
        data = json.loads(_json_part(raw))
    except json.JSONDecodeError as exc:
        raise ValueError(f"answer is not valid JSON: {exc}") from exc
    if not isinstance(data, dict):
        raise ValueError("answer must be a JSON object")
    data.pop("source", None)
    try:
        decision = Decision(**data)
    except ValidationError as exc:
        raise ValueError(f"answer does not match the schema: {exc.errors(include_url=False)}") from exc
    by_name = {m.name: m for m in menu}
    item = by_name.get(decision.action)
    if item is None:
        raise ValueError(f"unknown action {decision.action!r}; allowed: {sorted(by_name)}")
    if decision.level != item.level:
        raise ValueError(f"action {item.name!r} has level {item.level}, not {decision.level}")
    missing = [p for p in item.required_params if p not in decision.params]
    if missing:
        raise ValueError(f"action {item.name!r} is missing params {missing}")
    if not decision.success_check:
        decision.success_check = item.success_check
    return decision


# --------------------------------------------------------------------------
# decide()
# --------------------------------------------------------------------------
def decide(
    agent: Any,
    context: dict,
    menu: Iterable[Any],
    fallback: Decision | dict,
    *,
    incident_id: str | None = None,
    client: Any | None = None,
    config: dict | None = None,
) -> Decision:
    cfg = {**load_config(), **(config or {})}
    if config and "provider" in config and "api_key_env" not in config:
        # another provider than agents.yaml's: never send that file's key (e.g. a Groq key to NVIDIA)
        cfg["api_key_env"] = PROVIDER_DEFAULTS.get(config["provider"], {}).get("api_key_env")
    items = [MenuItem.coerce(m) for m in menu]
    rule = fallback if isinstance(fallback, Decision) else Decision(**fallback)
    agent_name = getattr(agent, "name", None) or str(agent)
    provider = cfg.get("provider", "fake")

    if provider == "off":
        return rule.model_copy(update={"source": "rules"})

    if client is None:
        factory = PROVIDERS.get(provider)
        if factory is None:
            client = ApiLLM()  # raises a clear error, logged below
        else:
            client = factory()
    if isinstance(client, FakeLLM):  # echo this call's rule decision once the script runs out
        client.fallback = rule.model_dump(exclude={"source"})

    messages = build_messages(agent, context, items)
    last_error = None
    for attempt in (1, 2):
        start = time.monotonic()
        raw = tokens_in = tokens_out = None
        try:
            raw, tokens_in, tokens_out = client.chat(messages, cfg)
        except Exception as exc:  # provider down, timeout, bad config
            _log_call(
                CallRecord(_id(), agent_name, incident_id, attempt, messages[-1]["content"], None, None, "error",
                           f"{type(exc).__name__}: {exc}", None, None, _ms(start), provider, cfg.get("model")),
                cfg.get("db_url"),
            )
            return rule.model_copy(update={"source": "llm_error"})
        try:
            decision = validate(raw, items)
        except ValueError as exc:
            last_error = str(exc)
            _log_call(
                CallRecord(_id(), agent_name, incident_id, attempt, messages[-1]["content"], raw, None, "invalid",
                           last_error, tokens_in, tokens_out, _ms(start), provider, cfg.get("model")),
                cfg.get("db_url"),
            )
            messages = messages + [
                {"role": "assistant", "content": raw},
                {"role": "user", "content": f"Your answer was rejected: {last_error}. Answer again with one valid JSON object."},
            ]
            continue
        _log_call(
            CallRecord(_id(), agent_name, incident_id, attempt, messages[-1]["content"], raw, decision.model_dump(), "ok",
                       None, tokens_in, tokens_out, _ms(start), provider, cfg.get("model")),
            cfg.get("db_url"),
        )
        return decision.model_copy(update={"source": "llm"})

    return rule.model_copy(update={"source": "llm_invalid", "reason": f"{rule.reason} [llm_invalid: {last_error}]".strip()})


def _id() -> str:
    return "llm-" + uuid.uuid4().hex[:12]


def _ms(start: float) -> int:
    return int((time.monotonic() - start) * 1000)

"""
Standard envelope definition for board messages using Pydantic.
Following Roadmap section 5.5.
"""
from datetime import datetime, timezone
from enum import Enum
from typing import Any, Dict, List, Optional
import uuid
from pydantic import BaseModel, Field


class MessageType(str, Enum):
    FINDING = "finding"
    QUESTION = "question"
    ANSWER = "answer"
    REQUEST = "request"
    CLAIM = "claim"


class MessageStatus(str, Enum):
    OPEN = "open"
    ANSWERED = "answered"
    DONE = "done"
    EXPIRED = "expired"


def generate_msg_id() -> str:
    """Generates an id formatted as msg- + 12 hex characters."""
    return f"msg-{uuid.uuid4().hex[:12]}"


class Message(BaseModel):
    id: str = Field(default_factory=generate_msg_id)
    time: str = Field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    type: MessageType
    kind: str
    sender: str
    to: List[str] = Field(default_factory=list)
    topic: str
    incident_id: Optional[str] = None
    reply_to: Optional[str] = None
    level: int = Field(ge=0, le=3)
    confidence: Optional[float] = Field(default=None, ge=0.0, le=1.0)
    summary: str = Field(max_length=160)
    evidence: Dict[str, Any] = Field(default_factory=dict)
    payload: Dict[str, Any] = Field(default_factory=dict)
    status: MessageStatus = Field(default=MessageStatus.OPEN)
    expires_at: Optional[str] = None
    run_id: str = "dev"

def _now() -> str:
    try:
        from shared import clock

        return clock.now().isoformat()
    except Exception:
        return datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def new_message(**fields: Any) -> dict:
    msg = {
        "id": "msg-" + uuid.uuid4().hex[:12],
        "time": _now(),
        "to": [],
        "incident_id": None,
        "reply_to": None,
        "level": 0,
        "confidence": None,
        "evidence": {},
        "payload": {},
        "status": "open",
        "expires_at": None,
        "run_id": "dev",
        **fields,
    }
    missing = [k for k in REQUIRED if k not in msg]
    if missing:
        raise ValueError(f"message is missing {missing}")
    if msg["type"] not in TYPES:
        raise ValueError(f"bad message type {msg['type']!r}")
    if len(msg["summary"]) > 160:
        raise ValueError("summary is longer than 160 characters")
    from shared.messages import Message  # the team's envelope: fail here, not on the real board

    Message(**{k: v for k, v in msg.items() if k in Message.model_fields})
    return msg


class InMemoryBoard:
    def __init__(self, subscriptions: dict[str, list[str]] | None = None):
        self.messages: list[dict] = []
        self.cursors: dict[str, int] = {}
        self.subscriptions = subscriptions or {}
        self.heartbeats: dict[str, str] = {}

    def post(self, msg: dict) -> str:
        msg = new_message(**msg)
        self.messages.append(msg)
        return msg["id"]

    def _for(self, agent: str, msg: dict) -> bool:
        if msg["sender"] == agent:
            return False
        if msg["to"]:
            return agent in msg["to"]
        return any(fnmatch.fnmatch(msg["topic"], pat) for pat in self.subscriptions.get(agent, []))

    def read(self, agent: str) -> list[dict]:
        start = self.cursors.get(agent, 0)
        self.cursors[agent] = len(self.messages)
        return [m for m in self.messages[start:] if self._for(agent, m)]

    def answer(self, original: dict, **fields: Any) -> str:
        original["status"] = "answered"
        return self.post(
            {
                "type": "answer",
                "kind": original["kind"],
                "to": [original["sender"]],
                "topic": original["topic"],
                "incident_id": original.get("incident_id"),
                "reply_to": original["id"],
                "run_id": original.get("run_id", "dev"),
                **fields,
            }
        )

    def thread(self, incident_id: str) -> list[dict]:
        return [m for m in self.messages if m.get("incident_id") == incident_id]

    def heartbeat(self, agent: str) -> None:
        self.heartbeats[agent] = _now()

DEFAULT_SUBSCRIPTIONS = {

    "trainer": ["dataset:*", "model:*", "expected_change:*"],
    "deployer": ["model:*", "deployment:*", "expected_change:*"],

BOARD_SCHEMA = """
CREATE TABLE IF NOT EXISTS messages (
    seq INTEGER PRIMARY KEY AUTOINCREMENT, id TEXT UNIQUE NOT NULL, time TEXT, type TEXT, kind TEXT,
    sender TEXT, "to" TEXT, topic TEXT, incident_id TEXT, reply_to TEXT, level INTEGER, confidence REAL,
    summary TEXT, evidence TEXT, payload TEXT, status TEXT, expires_at TEXT, run_id TEXT);
CREATE TABLE IF NOT EXISTS cursors (agent TEXT PRIMARY KEY, last_seq INTEGER);
CREATE TABLE IF NOT EXISTS heartbeats (agent TEXT PRIMARY KEY, time TEXT);
"""
COLUMNS = ("id", "time", "type", "kind", "sender", "to", "topic", "incident_id", "reply_to", "level", "confidence",
           "summary", "evidence", "payload", "status", "expires_at", "run_id")
JSON_COLUMNS = ("to", "evidence", "payload")


def load_subscriptions(path=None) -> dict[str, list[str]]:
    """agents.<name>.subscriptions from agents.yaml, else the roadmap defaults."""
    import os
    from pathlib import Path

    subs = {k: list(v) for k, v in DEFAULT_SUBSCRIPTIONS.items()}
    path = Path(path or os.environ.get("AGENTS_YAML", Path(__file__).resolve().parents[1] / "agents.yaml"))
    if path.exists():
        import yaml

        for name, cfg in ((yaml.safe_load(path.read_text(encoding="utf-8")) or {}).get("agents") or {}).items():
            if (cfg or {}).get("subscriptions"):
                subs[name] = list(cfg["subscriptions"])
    return subs


class SqliteBoard(InMemoryBoard):
    """InMemoryBoard's API on the messages, cursors and heartbeats tables of board.db.

    Every agent process opens the same file; read() keeps each agent's place in
    cursors, so a restarted agent carries on where it stopped.
    """

    def __init__(self, db_path, subscriptions: dict[str, list[str]] | None = None):
        self.db_path = str(db_path)
        self.subscriptions = subscriptions if subscriptions is not None else load_subscriptions()
        with self._connect() as con:
            con.executescript(BOARD_SCHEMA)

    def _connect(self):
        con = sqlite3.connect(self.db_path, timeout=10)
        con.execute("PRAGMA journal_mode=WAL")
        return _Closing(con)

    @staticmethod
    def _row(row: sqlite3.Row) -> dict:
        msg = {k: row[k] for k in COLUMNS}
        for k in JSON_COLUMNS:
            msg[k] = json.loads(msg[k]) if msg[k] else ([] if k == "to" else {})
        return msg

    def insert(self, con: sqlite3.Connection, msg: dict) -> str:
        """Validate and write one message on an open connection (lets a caller post in its own transaction)."""
        msg = new_message(**msg)
        values = [json.dumps(msg[k]) if k in JSON_COLUMNS else msg[k] for k in COLUMNS]
        cols = ",".join(f'"{c}"' for c in COLUMNS)
        con.execute(f"INSERT INTO messages ({cols}) VALUES ({','.join('?' * len(COLUMNS))})", values)
        return msg["id"]

    def post(self, msg: dict) -> str:
        with self._connect() as con:
            return self.insert(con, msg)

    def _select(self, where: str = "", params: tuple = ()) -> list[sqlite3.Row]:
        with self._connect() as con:
            con.row_factory = sqlite3.Row
            return con.execute(f"SELECT * FROM messages {where} ORDER BY seq", params).fetchall()

    @property
    def messages(self) -> list[dict]:
        return [self._row(r) for r in self._select()]

    def read(self, agent: str) -> list[dict]:
        with self._connect() as con:
            con.row_factory = sqlite3.Row
            cur = con.execute("SELECT last_seq FROM cursors WHERE agent = ?", (agent,)).fetchone()
            rows = con.execute("SELECT * FROM messages WHERE seq > ? ORDER BY seq", (cur[0] if cur else 0,)).fetchall()
            if rows:
                con.execute("INSERT INTO cursors VALUES (?, ?) ON CONFLICT(agent) DO UPDATE SET last_seq = excluded.last_seq",
                            (agent, rows[-1]["seq"]))
        return [m for m in map(self._row, rows) if self._for(agent, m)]

    def answer(self, original: dict, **fields: Any) -> str:
        msg_id = super().answer(original, **fields)  # builds the answer and posts it through self.post
        with self._connect() as con:
            con.execute("UPDATE messages SET status = 'answered' WHERE id = ?", (original["id"],))
        return msg_id

    def ask(self, msg: dict, timeout_s: float = 60, poll_s: float = 0.5) -> dict | None:
        """Post a question or request and wait for its answer; None when nobody answers in time."""
        msg_id = self.post(msg)
        deadline = time.monotonic() + timeout_s
        while time.monotonic() < deadline:
            rows = self._select("WHERE reply_to = ?", (msg_id,))
            if rows:
                return self._row(rows[0])
            time.sleep(poll_s)
        return None

    def thread(self, incident_id: str) -> list[dict]:
        return [self._row(r) for r in self._select("WHERE incident_id = ?", (incident_id,))]

    def heartbeat(self, agent: str) -> None:
        with self._connect() as con:
            con.execute("INSERT INTO heartbeats VALUES (?, ?) ON CONFLICT(agent) DO UPDATE SET time = excluded.time",
                        (agent, _now()))

    @property
    def heartbeats(self) -> dict[str, str]:
        with self._connect() as con:
            return dict(con.execute("SELECT agent, time FROM heartbeats").fetchall())


class _Closing:
    """`with` that commits (or rolls back) and then closes: on Windows an open board.db can't be deleted."""

    def __init__(self, con: sqlite3.Connection):
        self.con = con

    def __enter__(self) -> sqlite3.Connection:
        return self.con

    def __exit__(self, exc_type, *_):
        try:
            if exc_type is None:
                self.con.commit()
            else:
                self.con.rollback()
        finally:
            self.con.close()


def tail(db_path, follow: bool = True, poll_s: float = 1.0) -> None:
    """Print the board as it fills: python -m shared.standins tail board.db"""
    board, last = SqliteBoard(db_path, {}), 0
    while True:
        for row in board._select("WHERE seq > ?", (last,)):
            last = row["seq"]
            to = ",".join(json.loads(row["to"]) or ["*"])
            print(f"{row['time']}  {row['sender']:>8} -> {to:<14} {row['type']:8} {row['kind']:26} {row['summary']}", flush=True)
        if not follow:
            return
        time.sleep(poll_s)


class BaseAgent:
    """Message-handling part of the roadmap 5.8 loop. The full loop is P3's."""

    name = "base"
    menu: list = []

    def __init__(self, board: InMemoryBoard):
        self.board = board
        self.handled: set[str] = set()

    def handle(self, msg: dict) -> None:  # each agent overrides
        pass

    def safe_handle(self, msg: dict) -> None:
        if msg["id"] in self.handled:  # handlers must be idempotent
            return
        self.handled.add(msg["id"])
        try:
            self.handle(msg)
        except Exception as exc:  # never crash on a bad message
            print(f"[{self.name}] failed to handle {msg.get('kind')} {msg.get('id')}: {exc}")

    def step(self) -> None:
        for msg in self.board.read(self.name):
            self.safe_handle(msg)
        self.board.heartbeat(self.name)

===== shared/tables.py =====
"""
Shared tables in board.db (roadmap 5.11). Messages announce, tables hold state:
a restarted agent catches up by reading these tables.

This file holds the tables of the data path (owner P3): batches, batch_stats,
labels and faults. predictions and release_state (P5) can be added here the
same way.

When DRY_RUN=1 (inside a dry-run job) every write raises, so a dry run can
never change the real tables.
"""
from __future__ import annotations

import json
import os
import sqlite3
from typing import Iterable, Optional

import pandas as pd

from shared import clock

# `path` is the file to train on: the original batch, or the repaired copy once the Checker
# repaired it (the original then stays in `raw_path`). The Trainer reads `path` only.
BATCH_COLUMNS = {"batch_id": "TEXT PRIMARY KEY", "path": "TEXT", "status": "TEXT",
                 "rows": "INTEGER", "created_at": "TEXT", "checked_at": "TEXT", "reason": "TEXT",
                 "clean_path": "TEXT", "raw_path": "TEXT", "schema_version": "INTEGER",
                 "findings": "TEXT"}

SCHEMA = f"""
CREATE TABLE IF NOT EXISTS batches (
    {", ".join(f"{name} {kind}" for name, kind in BATCH_COLUMNS.items())});
CREATE TABLE IF NOT EXISTS batch_stats (
    batch_id TEXT, column_name TEXT, null_rate REAL, mean REAL, median REAL,
    p1 REAL, p99 REAL, PRIMARY KEY (batch_id, column_name));
CREATE TABLE IF NOT EXISTS labels (
    wine_id INTEGER PRIMARY KEY, quality REAL, available_at TEXT);
CREATE TABLE IF NOT EXISTS claims (
    incident_id TEXT, node TEXT, agent TEXT, claimed_at TEXT);
CREATE TABLE IF NOT EXISTS faults (
    fault_id TEXT, layer TEXT, target TEXT, injected_at TEXT, first_batch TEXT,
    expected TEXT, run_id TEXT);
"""

STATUSES = ("pending", "passed", "repaired", "quarantined")


class DryRunWriteError(RuntimeError):
    pass


class Tables:
    def __init__(self, path: str = "board.db"):
        self.db = sqlite3.connect(path, check_same_thread=False)
        self.db.row_factory = sqlite3.Row
        if path != ":memory:":
            self.db.execute("PRAGMA journal_mode=WAL")   # six agents writing at once
        self.db.executescript(SCHEMA)
        self._add_missing_batch_columns()

    def _add_missing_batch_columns(self) -> None:
        """The Trainer creates a smaller batches table (agents/trainer/data.py) when it starts
        first; add our columns to it instead of failing on the first write."""
        existing = {row[1] for row in self.db.execute("PRAGMA table_info(batches)")}
        for name, kind in BATCH_COLUMNS.items():
            if name not in existing:
                self.db.execute(f"ALTER TABLE batches ADD COLUMN {name} {kind.split()[0]}")
        self.db.commit()

    def _write(self, sql: str, params: Iterable = ()) -> None:
        if os.environ.get("DRY_RUN") == "1":
            raise DryRunWriteError("tables are read-only inside a dry run")
        self.db.execute(sql, tuple(params))
        self.db.commit()

    # ---- batches ----------------------------------------------------------------
    def add_batch(self, batch_id: str, path: str, rows: int) -> None:
        self._write("INSERT OR REPLACE INTO batches (batch_id, path, status, rows, created_at) "
                    "VALUES (?, ?, 'pending', ?, ?)", (batch_id, path, rows, clock.iso()))

    def set_batch_status(self, batch_id: str, status: str, reason: str = "",
                         clean_path: Optional[str] = None, schema_version: int = 1,
                         findings: Optional[list] = None) -> None:
        if status not in STATUSES:
            raise ValueError(f"unknown batch status {status!r}")
        self._write("UPDATE batches SET status=?, checked_at=?, reason=?, clean_path=?, "
                    "schema_version=?, findings=? WHERE batch_id=?",
                    (status, clock.iso(), reason, clean_path, schema_version,
                     json.dumps(findings or []), batch_id))

    def use_repaired_file(self, batch_id: str, repaired_path: str) -> None:
        """Training must use the repaired copy: `path` points to it, the original is kept
        in `raw_path`."""
        self._write("UPDATE batches SET raw_path=COALESCE(raw_path, path), path=? WHERE batch_id=?",
                    (repaired_path, batch_id))

    def batch(self, batch_id: str) -> Optional[dict]:
        row = self.db.execute("SELECT * FROM batches WHERE batch_id=?", (batch_id,)).fetchone()
        return dict(row) if row else None

    def batches(self, status: Optional[str] = None) -> list[dict]:
        if status:
            cur = self.db.execute("SELECT * FROM batches WHERE status=? ORDER BY batch_id", (status,))
        else:
            cur = self.db.execute("SELECT * FROM batches ORDER BY batch_id")
        return [dict(r) for r in cur]

    def pending_batches(self) -> list[dict]:
        return self.batches("pending")

    # ---- batch_stats ------------------------------------------------------------
    def add_batch_stats(self, batch_id: str, stats: list[dict]) -> None:
        for s in stats:
            self._write("INSERT OR REPLACE INTO batch_stats VALUES (?, ?, ?, ?, ?, ?, ?)",
                        (batch_id, s["column"], s["null_rate"], s["mean"], s["median"],
                         s["p1"], s["p99"]))

    def batch_stats(self) -> pd.DataFrame:
        return pd.read_sql_query("SELECT * FROM batch_stats ORDER BY batch_id", self.db)

    # ---- labels -----------------------------------------------------------------
    def add_labels(self, rows: list[tuple[int, float, str]]) -> None:
        for wine_id, quality, available_at in rows:
            self._write("INSERT OR REPLACE INTO labels VALUES (?, ?, ?)",
                        (int(wine_id), float(quality), available_at))

    def labels_available(self) -> pd.DataFrame:
        return pd.read_sql_query("SELECT * FROM labels WHERE available_at <= ?", self.db,
                                 params=(clock.iso(),))

    # ---- claims: which agent owns which incident, kept across restarts --------
    def record_claim(self, incident_id: str, node: str, agent: str) -> None:
        self._write("INSERT INTO claims VALUES (?, ?, ?, ?)", (incident_id, node, agent, clock.iso()))

    def recent_claim(self, nodes: list[str], since: str) -> Optional[str]:
        """Incident id of a claim on one of these nodes made at or after `since`, else None."""
        if not nodes:
            return None
        marks = ",".join("?" * len(nodes))
        row = self.db.execute(f"SELECT incident_id FROM claims WHERE node IN ({marks}) AND claimed_at >= ? "
                              "ORDER BY claimed_at DESC LIMIT 1", (*nodes, since)).fetchone()
        return row[0] if row else None

    # ---- faults (harness only; agents never read this table) ------------------
    def record_fault(self, fault_id: str, layer: str, target: dict, first_batch: str,
                     expected: dict, run_id: str = "dev") -> None:
        self._write("INSERT INTO faults VALUES (?, ?, ?, ?, ?, ?, ?)",
                    (fault_id, layer, json.dumps(target), clock.iso(), first_batch,
                     json.dumps(expected), run_id))

    def faults(self) -> list[dict]:
        return [dict(r) for r in self.db.execute("SELECT * FROM faults")]

"""

import uuid
from dataclasses import dataclass, field

class InMemoryBoard:
    def __init__(self):
        self.messages: list[Message] = []
        self.cursors: dict[str, int] = {}
        self.subscriptions: dict[str, list[str]] = {}
        self.last_seen: dict[str, datetime] = {}

    def subscribe(self, agent: str, patterns: list[str]) -> None:
        self.subscriptions[agent] = list(patterns)

    def post(self, msg) -> str:
        from shared.base_agent import as_message
        msg = as_message(msg)
        self.messages.append(msg)
        return msg.id

    def read(self, agent: str) -> list[Message]:
        """New messages named for this agent, or broadcast on a subscribed topic."""
        start = self.cursors.get(agent, 0)
        self.cursors[agent] = len(self.messages)
        patterns = self.subscriptions.get(agent, [])
        return [m for m in self.messages[start:] if m.sender != agent and (
            agent in m.to or (not m.to and any(fnmatch.fnmatch(m.topic, p) for p in patterns)))]

    def answer(self, original, **fields) -> str:
        if isinstance(original, dict):           # find the stored message to mark it answered
            original = next(m for m in self.messages if m.id == original["id"])
        original.status = MessageStatus.ANSWERED
        fields.setdefault("summary", f"answer to {original.kind}")
        fields.setdefault("level", original.level)
        fields.setdefault("time", clock.iso())
        msg = Message(type=MessageType.ANSWER, kind=original.kind, to=[original.sender],
                      topic=original.topic, reply_to=original.id,
                      incident_id=original.incident_id, **fields)
        return self.post(msg)

    def ask(self, msg: Message, timeout_s: float = 0) -> Optional[Message]:
        """Posts the question; returns an answer already on the board, else None."""
        self.post(msg)
        return next((m for m in self.messages if m.reply_to == msg.id), None)

    def thread(self, incident_id: str) -> list[Message]:
        return [m for m in self.messages if m.incident_id == incident_id]

    def heartbeat(self, agent: str) -> None:
        self.last_seen[agent] = clock.now()

    def by_kind(self, kind: str) -> list[Message]:
        return [m for m in self.messages if m.kind == kind]


class FakeLLM:
    """Returns scripted decisions in order, then the fallback. Records every call."""

    def __init__(self, script: Optional[list] = None):
        self.script = list(script or [])
        self.calls: list[dict] = []

    def decide(self, agent, context, menu, fallback):
        self.calls.append({"agent": agent.name, "context": context})
        return self.script.pop(0) if self.script else fallback

class InMemoryMemory:
    def __init__(self):
        self.rows: list[dict] = []

    def remember(self, agent, incident_id, problem, evidence, action, params, level,
                 outcome, time_to_recover_s) -> str:
        mem_id = f"mem-{len(self.rows) + 1:04d}"
        self.rows.append(dict(id=mem_id, agent=agent, incident_id=incident_id, problem=problem,
                              evidence=evidence, action=action, params=params, level=level,
                              outcome=outcome, time_to_recover_s=time_to_recover_s))
        return mem_id

    def similar(self, problem: str, agent: Optional[str] = None, k: int = 5) -> list[dict]:
        words = set(problem.lower().split())
        scored = [(len(words & set(r["problem"].lower().split())), r) for r in self.rows
                  if agent is None or r["agent"] == agent]
        return [r for s, r in sorted(scored, key=lambda x: -x[0]) if s > 0][:k]


class SimpleIncidents:
    """Joins an open incident that shares a node within the window, else opens one."""

    def __init__(self):
        self.open: list[dict] = []

    def open_or_join(self, nodes: list[str], window_s: float) -> str:
        now = clock.now()
        for inc in self.open:
            if set(nodes) & inc["nodes"] and now - inc["opened_at"] <= timedelta(seconds=window_s):
                inc["nodes"] |= set(nodes)
                return inc["id"]
        inc = {"id": "inc-" + uuid.uuid4().hex[:8], "nodes": set(nodes), "opened_at": now}
        self.open.append(inc)
        return inc["id"]
